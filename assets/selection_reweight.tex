\begin{algorithm}[H]
    \caption{Reinforced Flow Matching}
    \label{alg:velocity-add-delete}
    \textbf{Require:} Initial model $\theta_0$, reward function $R$, prompt distribution $p(c)$, number of prompts per iteration $C$, rollout size per prompt $K$, batch size $B$, total samples per iteration $D = KC$, rollout coverage coefficient $\beta$, mass-shift clip $\Delta_{\mathrm{clip}}$, reference regularization $\lambda_{\mathrm{ref}}$, EMA rate $\eta$, reward-scale constant $\epsilon_R$, numerical constant $\epsilon_{\mathrm{alg}}$, gradient accumulation steps $N_{\mathrm{acc}}$.\\
    \textbf{Initialize:} Training policy $\theta \leftarrow \theta_0$, old policy $\theta_{\mathrm{old}} \leftarrow \theta_0$, reference policy $\theta_{\mathrm{ref}} \leftarrow \theta_0$, data buffer $\mathcal{D} \leftarrow \emptyset$.
    \begin{algorithmic}[1]
        \For{\text{each iteration}}
            \For{\text{each prompt $c_j \sim p(c)$, $j = 1, \dots, C$}} \Comment{Rollout Step, Data Collection}
                \State Sample $K$ clean images $\{\mathbf{x}_0^{j,k}\}_{k=1}^{K}$ with $\theta$ conditioned on $c_j$.
                \State Evaluate rewards $R_{j,k} \leftarrow R(\mathbf{x}_0^{j,k}, c_j)$ for $k = 1, \dots, K$.
            \EndFor

            \State Compute prompt-wise reward means
            $\bar{R}_{c_j} \leftarrow \frac{1}{K}\sum_{k=1}^{K} R_{j,k}$ for $j = 1, \dots, C$.

            \State Compute global reward mean
            $\bar{R} \leftarrow \frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K} R_{j,k}$.

            \State Compute global reward scale
            $\sigma_R \leftarrow \sqrt{\frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K}(R_{j,k} - \bar{R})^2}.$

            \For{\text{each prompt $c_j$, $j = 1, \dots, C$}}
                \State Compute normalized weights
                \[
                    \Delta_{j,k} \leftarrow \frac{1}{\Delta_{\mathrm{clip}}}
                    \operatorname{clip}\!\left(
                        \frac{R_{j,k} - \bar{R}_{c_j}}{\sigma_R},
                        -\Delta_{\mathrm{clip}},
                        \Delta_{\mathrm{clip}}
                    \right),
                    \quad
                    w_{j,k}-1\leftarrow \Delta_{j,k},
                    \quad k = 1, \dots, K.
                \]

                \State Store
                $\mathcal{D}\leftarrow\mathcal{D}\cup
                \{(c_j,\mathbf{x}_0^{j,k},w_{j,k})\}_{k=1}^{K}$.
            \EndFor

            \For{\text{each mini batch }
            $\{(c_i,\mathbf{x}_i^0,w_i)\}_{i=1}^{B}\subset\mathcal{D}$}
                \State Sample $t_i\sim\mathcal{U}(0,1)$,
                $\boldsymbol{\epsilon}_i\sim\mathcal{N}(\mathbf{0},\mathbf{I})$;
                set $\mathbf{x}_i^{t_i}\leftarrow
                (1-t_i)\mathbf{x}_i^0+t_i\boldsymbol{\epsilon}_i$.
                \State Predict $\mathbf{x}_i^\theta$,
                $\mathbf{x}_i^{\mathrm{old}}$, and $\mathbf{x}_i^{\mathrm{ref}}$
                from $(\mathbf{x}_i^{t_i},c_i,t_i)$ with
                $\theta$, $\theta_{\mathrm{old}}$, and $\theta_{\mathrm{ref}}$.
                \State $\mathbf{x}_i^{\mathrm{target}}\leftarrow
                \mathbf{x}_i^{\mathrm{old}}+
                \beta(w_i-1)(\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}})$.
                \State Compute
                \[
                    \begin{aligned}
                    \mathcal{L}_{\mathrm{policy}}
                    &\leftarrow\frac{1}{B}\sum_{i=1}^{B}
                    \frac{\|\mathbf{x}_i^\theta-\mathbf{x}_i^{\mathrm{target}}\|_2^2}
                    {\operatorname{sg}[\operatorname{mean}
                    |\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}}|]},\\
                    \mathcal{L}_{\mathrm{ref}}
                    &\leftarrow\frac{1}{B}\sum_{i=1}^{B}
                    \|\mathbf{v}_i^\theta-\mathbf{v}_i^{\mathrm{ref}}\|_2^2.
                    \end{aligned}
                \]
                \State Backpropagate
                $(\Delta_{clip}\mathcal{L}_{\mathrm{policy}}+
                \lambda_{\mathrm{ref}}\mathcal{L}_{\mathrm{ref}})/N_{\mathrm{acc}}$.
            \EndFor
            \State Update $\theta$; set
            $\theta_{\mathrm{old}}\leftarrow
            \eta\theta_{\mathrm{old}}+(1-\eta)\theta$ and
            $\mathcal{D}\leftarrow\emptyset$.
        \EndFor
    \end{algorithmic}
    \textbf{Output:} $\mathbf{v}_\theta$
\end{algorithm}


\begin{algorithm}[H]
    \scriptsize
    \setlength{\abovedisplayskip}{1pt}
    \setlength{\belowdisplayskip}{1pt}
    \renewcommand{\arraystretch}{0.9}
    \caption{Scheme B: Symmetric Soft Selection with Boost-Only Floor}
    \label{alg:x-prediction-symmetric-soft-selection}
    \textbf{Require:} Initial model $\theta_0$, reward function $R$,
    prompt distribution $p(c)$, $C$
    prompts per iteration, $K$ samples per prompt, $D=CK$, batch size $B$,
    retained ratio $\rho$, shift strength $\beta$, mass-shift clip
    $\Delta_{\mathrm{clip}}$, reference regularization $\lambda_{\mathrm{ref}}$,
    EMA rate $\eta$, accumulation steps $N_{\mathrm{acc}}$.\\
    \textbf{Initialize:}
    $\theta,\theta_{\mathrm{old}},\theta_{\mathrm{ref}}\leftarrow\theta_0$,
    $\mathcal{D}\leftarrow\emptyset$.
    \begin{algorithmic}[1]
        \For{\text{each iteration}}
            \For{\text{each prompt $c_j\sim p(c)$, $j=1,\dots,C$}}
                \State Sample $\{\mathbf{x}_0^{j,k}\}_{k=1}^{K}$ with $\theta$
                conditioned on $c_j$ and set
                $R_{j,k}\leftarrow R(\mathbf{x}_0^{j,k},c_j)$.
            \EndFor

            \State Compute global reward mean
            $\bar{R}\leftarrow \frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K}R_{j,k}$.
            \State Compute global reward scale
            $\sigma_R \leftarrow \sqrt{\frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K}(R_{j,k} - \bar{R})^2}$.

            \For{\text{each prompt $c_j\sim p(c)$, $j=1,\dots,C$}}
                \State Set the mass-shift center to the prompt-group mean:
                $b_j\leftarrow K^{-1}\sum_{k=1}^{K}R_{j,k}$.
                \State Within this prompt group, compute
                \[
                    \Delta_{j,k}\leftarrow
                    \frac{1}{\Delta_{\mathrm{clip}}}
                    \operatorname{clip}\!\left(
                        \frac{R_{j,k}-b_j}{\sigma_R},
                        -\Delta_{\mathrm{clip}},
                        \Delta_{\mathrm{clip}}
                    \right),
                    \qquad k=1,\dots,K.
                \]
                \State Initialize $s_{j,k}\leftarrow0$ for $k=1,\dots,K$.

                \For{\text{each side $\tau\in\{-,+\}$}}
                    \State Define
                    $I_j^-\leftarrow\{k:\Delta_{j,k}<0\}$ and
                    $I_j^+\leftarrow\{k:\Delta_{j,k}>0\}$; use $I_j^\tau$.
                    \If{$I_j^\tau\neq\emptyset$}
                        \State Partition $I_j^\tau$ into exact equal-magnitude groups
                        $G_{j,1}^\tau,\dots,G_{j,m_j^\tau}^\tau$ such that
                        \[
                            G_{j,r}^\tau
                            =\{k\in I_j^\tau:|\Delta_{j,k}|=a_{j,r}^\tau\},
                            \qquad
                            a_{j,1}^\tau>\cdots>a_{j,m_j^\tau}^\tau>0.
                        \]
                        \State Compute group masses and the target retained mass:
                        \[
                            m_{j,r}^\tau\leftarrow
                            \sum_{k\in G_{j,r}^\tau}|\Delta_{j,k}|,
                            \qquad
                            T_j^\tau\leftarrow
                            \rho\sum_{r=1}^{m_j^\tau}m_{j,r}^\tau.
                        \]
                        \State Find the boundary group and mass before it:
                        \[
                            r_{j,\rho}^\tau\leftarrow
                            \min\left\{r:
                            \sum_{u=1}^{r}m_{j,u}^\tau\ge T_j^\tau\right\},
                            \qquad
                            P_j^\tau\leftarrow
                            \sum_{u=1}^{r_{j,\rho}^\tau-1}m_{j,u}^\tau.
                        \]
                        \State Assign one shared fraction to the complete boundary group:
                        \[
                            f_j^\tau\leftarrow
                            \frac{T_j^\tau-P_j^\tau}
                            {m_{j,r_{j,\rho}^\tau}^\tau}.
                        \]
                        \State Allocate the strong-tail mass without splitting a tie:
                        \[
                            s_{j,k}\leftarrow
                            \begin{cases}
                                \Delta_{j,k},
                                &k\in G_{j,r}^\tau,\ r<r_{j,\rho}^\tau,\\
                                f_j^\tau\Delta_{j,k},
                                &k\in G_{j,r_{j,\rho}^\tau}^\tau,\\
                                0,
                                &k\in G_{j,r}^\tau,\ r>r_{j,\rho}^\tau.
                            \end{cases}
                        \]
                    \EndIf
                \EndFor

                \State First finish the original selection and rescaling:
                \[
                    \widetilde{\Delta}_{j,k}\leftarrow s_{j,k}/\rho,
                    \qquad k=1,\dots,K.
                \]
            \EndFor

            \State After selection has finished for all groups, apply the
            boost-only floor to every sample:
            \[
                w_{j,k}-1\leftarrow
                \begin{cases}
                    \Delta_{j,k},
                    &|\widetilde{\Delta}_{j,k}|<|\Delta_{j,k}|,\\
                    \widetilde{\Delta}_{j,k},
                    &|\widetilde{\Delta}_{j,k}|\ge|\Delta_{j,k}|,
                \end{cases}
                \qquad j=1,\dots,C,\quad k=1,\dots,K.
            \]
            \State Hence $|w_{j,k}-1|\ge|\Delta_{j,k}|$ for every sample:
            attenuated candidates are restored to the original signal, while
            equal or amplified candidates are kept unchanged.
            \State Store
            $\mathcal{D}\leftarrow\mathcal{D}\cup
            \{(c_j,\mathbf{x}_0^{j,k},w_{j,k}):
            j=1,\dots,C,\ k=1,\dots,K\}$.

            \For{\text{each mini batch }
            $\{(c_i,\mathbf{x}_i^0,w_i)\}_{i=1}^{B}\subset\mathcal{D}$}
                \State Sample $t_i\sim\mathcal{U}(0,1)$,
                $\boldsymbol{\epsilon}_i\sim\mathcal{N}(\mathbf{0},\mathbf{I})$;
                set $\mathbf{x}_i^{t_i}\leftarrow
                (1-t_i)\mathbf{x}_i^0+t_i\boldsymbol{\epsilon}_i$.
                \State Predict $\mathbf{x}_i^\theta$,
                $\mathbf{x}_i^{\mathrm{old}}$, and $\mathbf{x}_i^{\mathrm{ref}}$
                from $(\mathbf{x}_i^{t_i},c_i,t_i)$ with
                $\theta$, $\theta_{\mathrm{old}}$, and $\theta_{\mathrm{ref}}$.
                \State $\mathbf{x}_i^{\mathrm{target}}\leftarrow
                \mathbf{x}_i^{\mathrm{old}}+
                \beta(w_i-1)(\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}})$.
                \State Compute
                \[
                    \begin{aligned}
                    \mathcal{L}_{\mathrm{policy}}
                    &\leftarrow\frac{1}{B}\sum_{i=1}^{B}
                    \frac{\|\mathbf{x}_i^\theta-\mathbf{x}_i^{\mathrm{target}}\|_2^2}
                    {\operatorname{sg}[\operatorname{mean}
                    |\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}}|]},\\
                    \mathcal{L}_{\mathrm{ref}}
                    &\leftarrow\frac{1}{B}\sum_{i=1}^{B}
                    \|\mathbf{v}_i^\theta-\mathbf{v}_i^{\mathrm{ref}}\|_2^2.
                    \end{aligned}
                \]
                \State Backpropagate
                $(\Delta_{\mathrm{clip}}\mathcal{L}_{\mathrm{policy}}+
                \lambda_{\mathrm{ref}}\mathcal{L}_{\mathrm{ref}})/N_{\mathrm{acc}}$.
            \EndFor
            \State Update $\theta$; set
            $\theta_{\mathrm{old}}\leftarrow
            \eta\theta_{\mathrm{old}}+(1-\eta)\theta$ and
            $\mathcal{D}\leftarrow\emptyset$.
        \EndFor
    \end{algorithmic}
    \textbf{Output:} $\mathbf{x}_\theta$
\end{algorithm}

                \State $\mathbf{x}_i^{\mathrm{target}}\leftarrow
                \mathbf{x}_i^{\mathrm{old}}+
                \beta(w_i-1)(\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}})$.
                \State Compute

All variants use the same normalized and clipped signal $\Delta_{j,k}$ as
the base scheme.  Scheme B selects the strong tail on both sides, whereas
Scheme C keeps all negative samples and selects only the positive strong
tail.  Before applying the boost-only floor, division by $\rho$ preserves
the selected side's mass (exactly for the tie-aware soft allocation and
approximately for hard selection).

The boost-only Scheme B then compares the fully selected candidate
$\widetilde{\Delta}_{j,k}$ with the original $\Delta_{j,k}$.  It restores
$\Delta_{j,k}$ whenever selection reduced the magnitude and retains
$\widetilde{\Delta}_{j,k}$ otherwise.  Consequently, its final absolute mass
is never smaller than the base mass and is strictly larger exactly when at
least one candidate is amplified; fixed-mass preservation no longer holds
after the floor.  When $\rho=1$, the candidate already equals
$\Delta_{j,k}$, so the method reduces exactly to the base scheme.

\paragraph{Baseline equivalence check.}
The run with $\rho=1$ and $\beta=1$ verifies that the selection
implementation reduces to the base scheme: no sample is discarded, hence
$w_{j,k}-1=\Delta_{j,k}$.  The expected result is that its learning curve and final metrics match the base scheme.

\paragraph{Scheme B: symmetric selection.}
 Comparisons between different $\rho$ at fixed $\beta$ test whether
concentrating both positive and negative signals on strong-tail samples is
better than retaining borderline samples; comparisons between different $\beta$ at fixed $\rho$ test
the required target-shift magnitude.  The intended conclusion is whether a
moderate $\rho$ and a compatible $\beta$ improve convergence or
stability over the base scheme, and whether aggressive selection must be
paired with a smaller $\beta$.

\paragraph{Scheme C: asymmetric selection.}
Comparing B and C at the same $(\rho,\beta)$ isolates this asymmetry.  Better and equally stable results from C would support keeping all negative samples and filtering only borderline positive samples; better results from B would favor symmetric selection, while a best result at $\rho=1$ would indicate no benefit from hard selection.

\paragraph{Problem with equal values.}
The original implementation sorts and selects individual samples.  If the
selection boundary falls among several samples with the same $\Delta_i$, it
may keep some of them and discard the others according only to their input
order.  Thus equally strong samples can receive different weights.  The
discrete boundary also means that the retained mass is generally only close
to, rather than exactly, $\rho\sum_i|\Delta_i|$.

\paragraph{Equal-value correction.}
The correction implementation first groups samples with exactly equal $|\Delta_i|$.  Starting from the
strongest groups, it keeps complete groups until the next group would exceed
the target mass $\rho\sum_i|\Delta_i|$.  Every sample in this boundary group
then receives the same fractional multiplier, chosen so that the remaining
required mass is shared equally across the group.  

For example, consider the
positive signals $\{1.0,0.5,0.5,0.5,0.25,0.25\}$ with $\rho=1/2$.  Their total
mass is $3.0$, so the target retained mass is $1.5$.  The strongest group
$\{1.0\}$ is completely retained, leaving $0.5$ mass to allocate.  The next
equal-value group is $\{0.5,0.5,0.5\}$, so this remaining mass is shared equally: each
sample receives the pre-rescaling signal $0.5/3=1/6$.  The weakest group
$\{0.25,0.25\}$ is completely discarded.  Thus the pre-rescaling signals are
$\{1.0,1/6,1/6,1/6,0,0\}$, whose mass is exactly $1.5$.  After division by
$\rho=1/2$, the candidates $\widetilde{\Delta}_i$ are
$\{2.0,1/3,1/3,1/3,0,0\}$.  The boost-only floor restores every candidate
weaker than its original signal, producing the final $w_i-1$ values
$\{2.0,0.5,0.5,0.5,0.25,0.25\}$.  Their total mass is $4.0$: only the
strongest sample is enhanced, and no sample is weakened.  Scheme B applies
this procedure to both sides, whereas Scheme C keeps all negative samples
and applies selection only to the positive side.


\paragraph{Equal-value validation experiments.}
We evaluate the effect of equal-value groups with two sets of single-reward
experiments:

First, using the original implementation, we run both Schemes B
and C on PickScore with $\beta=1.0$ and
$\rho\in\{1/3,1/2,2/3\}$.  PickScore produces continuous rewards, so exactly
equal rewards are much rarer than with GenEval; these runs measure hard
selection when its result is largely unaffected by input-order selection
among equal values.  

Second, using the equal-value implementation, we run Schemes B and C on GenEval with $\beta=1.0$ and
$\rho=1/2$.  Comparing these runs with the corresponding original GenEval
runs at the same $(\rho,\beta)$ isolates the effect of treating equal-reward
samples consistently and preserving the selected mass exactly.
