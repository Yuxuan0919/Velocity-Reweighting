\begin{abstract}
Large language model with reinforcement learning
usually learns a reward-tilted conditional distribution,
in a parametric form.
In diffusion model,
the distribution for the clean data 
is non-parametric, and
represented by samples,
and the diffusion model learns a denoising process
to approximate the distribution.
Inspired by this,
we decompose the problem into two steps.
reweight the samples according to rewards,
and update the flow matching with weighted samples
in an incremental way.
The reweighting scheme is a non-parametric way for the reward-tilted distribution,
formulated
from KL-constrained reward maximization. 
The incremental update consists of the target velocity 
estimation from the original velocity
and velocity correction that depends on  mass shift 
and timestep-dependent contribution weight,
followed by an MSE loss for velocity update.
We also note that
several existing methods
can also be casted into 
sampling reweighting and 
velocity correction.


\end{abstract}

\subsection{Reward-Tilted Sample Reweighting}
\noindent\textbf{KL-constrained reward maximization.}
We first consider a KL-constrained reward maximization objective, which
updates the generative distribution toward high-reward regions while
preserving proximity to the original distribution. Given an old distribution
$\pi_{\mathrm{old}}(\mathbf{z})$
and the reward $r(\mathbf{z})$, we solve
\begin{align}
\max_{\pi}
\quad
\mathbb{E}_{\mathbf{z}\sim\pi}
[r(\mathbf{z})]
-
\gamma
D_{\mathrm{KL}}
(\pi(\mathbf{z})\|
\pi^{\mathrm{o}}(\mathbf{z})).
\end{align}
The optimal solution can be obtained by exponential tilting:
\begin{align}
\pi^{\mathrm{n}}(\mathbf{z})
=
\frac{1}{Z}
\pi^o(\mathbf{z})
\exp
\left(
\frac{r(\mathbf{z})}{\gamma}
\right),\label{eqn:exponentialtilting}
\end{align}
where $Z$ is the normalization constant. Therefore, the KL-constrained reward optimization
can be interpreted as reweighting the original distribution according to the
exponential reward preference.

\noindent\textbf{Sample reweighting.}
We transform the 
reward-tilted distribution
from the parametric form
to the non-parametric form.
We denote $\mathbf{z}$,
a sample drawn from
the old distribution.
Its reward-induced unnormalized weight is
$\exp
\left(
\frac{r(\mathbf z)}{\gamma}
\right)$.
The normalization constant is 
\begin{align}
Z = \mathbb{E}_{\mathbf{z} \sim \pi^o}\left[\exp
\left(
\frac{r(\mathbf z)}{\gamma}
\right)\right].
\end{align}
Thus,
the normalized weight becomes
\begin{align}
w(\mathbf z) =
\frac{1}{Z}
\exp
\left(
\frac{r(\mathbf z)}{\gamma}
\right)
=
\frac{
\exp(r(\mathbf z)/\gamma)
}
{
\frac{1}{D}
\sum_j
\exp(r(\mathbf z_j)/\gamma)
}.    
\end{align}

In summary,
the reward-tilted distribution 
is represented
by reweighted rollout samples:
$\{(\mathbf{z}, w(\mathbf{z})\}$





\emph{Partial rollout coverage.}
In practice, rollout samples may only cover a subset of the original 
distribution. 
We model the old distribution as a mixture of an unobserved
background region and an observed rollout region:
\begin{align}
\pi^{\mathrm{o}}(\mathbf z)
=
(1-\beta)
\pi_{\mathrm{bg}}(\mathbf z)
+
\beta
\pi_{\mathcal D}(\mathbf z),
\end{align}
where $\beta$ denotes the probability mass covered by rollout samples,
$\pi_{\mathcal D}$ is the empirical rollout distribution, and
$\pi_{\mathrm{bg}}$ represents the remaining undrawn population.

Sample reweighting can be decomposed according to the two
sub-populations.
For the background region, 
there is no reward available
and
we assume that unnormalized weight is $1$.
Thus, the normalization constant is
\begin{align}
    Z
    = (1-\beta)
+
\frac{\beta}{D}
\sum_i
\exp(r(\mathbf z_i)/\gamma),
\end{align}
where $\{\mathbf z_i\}$
are the $D$ rollout samples.

For the observed rollout region, each clean sample $\mathbf z$ receives an
additional reward-dependent correction:
\begin{align}
w(\mathbf z)
&=
\frac{
\beta
\exp(r(\mathbf z)/\gamma)
}
{
(1-\beta)
+
\frac{\beta}{D}
\sum_j
\exp(r(\mathbf z_j)/\gamma)
}.
\end{align}


\subsection{Incremental Velocity Prediction}
\noindent\textbf{Optimal velocity field.} 
We first give the optimal velocity field in standard flow matching. 
It is the posterior expectation of the conditional velocity:
\begin{align}
\mathbf{v}(\mathbf{x}_t,t)
=
\mathbb{E}_{\mathbf{z} | \mathbf{x}_t}
\left[
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\mid
\mathbf{x}_t
\right].
\end{align}
The distribution is
\begin{align}
p(\mathbf{z}|\mathbf{x}_t)
=
\frac{
\pi(\mathbf{z})
p_t(\mathbf{x}_t|\mathbf{z})
}
{
\int
\pi(\mathbf{z}')
p_t(\mathbf{x}_t|\mathbf{z}')
d\mathbf{z}'
} 
= 
\frac{
\pi(\mathbf{z})
p_t(\mathbf{x}_t|\mathbf{z})
}
{
p_t(\mathbf{x}_t)
},
\end{align}
where
$
p_t(\mathbf{x}_t)$
denotes the marginal distribution.
The optimal velocity field becomes
\begin{align}
\mathbf{v}(\mathbf{x}_t,t)
&= 
\mathbb{E}_{\mathbf{z}\sim\pi}
\left[
\frac{p_t(\mathbf{x}_t|\mathbf{z})}
{
p_t(\mathbf{x}_t)
}
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right]
\\
&=
\mathbb{E}_{\mathbf{z}\sim\pi}
[
\alpha_t(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
].
\end{align}
Here, $\alpha_t(\mathbf{x}_t,\mathbf{z})
=
\frac{
p_t(\mathbf{x}_t|\mathbf{z})
}
{
p_t(\mathbf{x}_t)
}$ (conditional-to-marginal ratio) 
serves as a posterior correction
factor that transforms the prior distribution $\pi(\mathbf{z})$ into the
conditional distribution $p(\mathbf{z}|\mathbf{x}_t)$.


\noindent\textbf{Velocity field
from unweighting to reweighting.} 
The optimal velocity field 
for reweighted samples
is given as the following,
\begin{align}
\mathbf{v}^{n}(\mathbf{x}_t,t)
&=
\mathbb{E}_{\mathbf{z}\sim\pi^n}
\left[
\alpha_t^n(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right]\\
&=
\mathbb{E}_{\mathbf{z}\sim\pi^{\mathrm{o}}}
\left[
\frac{\pi^n(\mathbf{z})}
{\pi^o(\mathbf{z})}
\alpha_t^n(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right]\\
&\approx
\mathbb{E}_{\mathbf{z}\sim\pi^{\mathrm{o}}}
\left[
\frac{\pi^n(\mathbf{z})}
{\pi^{\mathrm{o}}(\mathbf{z})}
\alpha_t^{\textrm{o}}(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right].
\end{align}

respect to $\mathbf{v}_{\theta}$. Importantly, the optimal solution is not a
simple expectation over $\pi^n(\mathbf{z})$, since the velocity field is
defined on the marginal distribution of $\mathbf{x}_t$. Instead, it is given
by the posterior expectation of the conditional velocity:
\begin{align}
\mathbf{v}^{n}(\mathbf{x}_t,t)
=
\mathbb{E}
\left[
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\mid
\mathbf{x}_t
\right].
\end{align}
Under the reward-optimized distribution $\pi^n(\mathbf{z})$, the posterior
distribution is
\begin{align}
p^n(\mathbf{z}|\mathbf{x}_t)
=
\frac{
\pi^n(\mathbf{z})
p_t(\mathbf{x}_t|\mathbf{z})
}
{
\int
\pi^n(\mathbf{z}')
p_t(\mathbf{x}_t|\mathbf{z}')
d\mathbf{z}'
} 
= 
\frac{
\pi^n(\mathbf{z})
p_t(\mathbf{x}_t|\mathbf{z})
}
{
p_t^n(\mathbf{x}_t)
},
\end{align}
where
$
p_t^n(\mathbf{x}_t)$
denotes the marginal distribution induced by the reward-optimized source
distribution.

Therefore, the reward-optimal velocity field becomes
\begin{align}
\mathbf{v}^{n}(\mathbf{x}_t,t)
&=
\frac{
\mathbb{E}_{\mathbf{z}\sim\pi^n}
\left[
p_t(\mathbf{x}_t|\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right]
}
{
p_t^n(\mathbf{x}_t)
}\\
&=
\mathbb{E}_{\mathbf{z}\sim\pi^n}
\left[
\frac{p_t(\mathbf{x}_t|\mathbf{z})}{p_t^n(\mathbf{x}_t)}
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right] \\
&=
\mathbb{E}_{\mathbf{z}\sim\pi^n}
[
\alpha_t^n(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
].
\end{align}
Here, $\alpha_t^n(\mathbf{x}_t,\mathbf{z})
=
\frac{
p_t(\mathbf{x}_t|\mathbf{z})
}
{
p_t^n(\mathbf{x}_t)
}$ (conditional-to-marginal ratio) 
serves as a posterior correction
factor that transforms the prior distribution $\pi^n(\mathbf{z})$ into the
conditional distribution $p^n(\mathbf{z}|\mathbf{x}_t)$.

This formulation shows that reward optimization modifies the flow dynamics
through posterior reweighting of conditional trajectories. Unlike simple
reward-weighted regression, which only changes the contribution of individual
samples, the proposed formulation explicitly characterizes the reward-optimal
velocity field by modifying the conditional aggregation of flow trajectories.

\noindent\textbf{Reward-tilted velocity field.}
The reward-tilted velocity field is obtained by replacing the original
latent distribution $\pi^{\mathrm{o}}(\mathbf{z})$ with the reward-improved
distribution $\pi^{\mathrm{n}}(\mathbf{z})$. Specifically, the optimal distribution
reweights the latent samples according to their rewards, leading to
\begin{align}
\mathbf{v}^{n}(\mathbf{x}_t,t)
&=
\mathbb{E}_{\mathbf{z}\sim\pi^n}
\left[
\alpha_t^n(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right]\\
&=
\mathbb{E}_{\mathbf{z}\sim\pi^{\mathrm{o}}}
\left[
\frac{\pi^n(\mathbf{z})}
{\pi_{\mathrm{old}}(\mathbf{z})}
\alpha_t^n(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right]\\
&\approx
\mathbb{E}_{\mathbf{z}\sim\pi^{\mathrm{o}}}
\left[
\frac{\pi^n(\mathbf{z})}
{\pi^{\mathrm{o}}(\mathbf{z})}
\alpha_t^{\textrm{o}}(\mathbf{x}_t,\mathbf{z})
\mathbf{v}_t(\mathbf{x}_t|\mathbf{z})
\right].
\end{align}
Here,
$\frac{\pi^n(\mathbf{z})}{\pi^{\mathrm{o}}(\mathbf{z})}$ is the importance
weight that transforms samples drawn from the original latent distribution
into samples from the reward-optimized distribution. Therefore, the
reward-guided velocity can be interpreted as an importance-weighted
aggregation of the original conditional velocity fields, where high-reward
latent variables receive larger contributions to the resulting vector field.

The resulting velocity update can be interpreted as an incremental correction
of the current flow field:
\begin{align}
\color{red}
\mathbf v^n(\mathbf x_t,t)
=
\mathbf v^{\mathrm{o}}(\mathbf x_t,t)
+
\mathbb E_{\mathbf z\sim P_{\mathcal D}}
[
\mathbf c_t(\mathbf x_t,\mathbf z)
].
\end{align}
The
sample-wise correction with adaptive weight is
\begin{align}
\mathbf c_t(\mathbf x_t,\mathbf z)
=
\underbrace{
\alpha^{\textrm{o}}_t(\mathbf x_t,\mathbf z)
}_{\text{trajectory contribution}}
\times
\underbrace{
\beta\left(
\exp(r(\mathbf z)/\gamma)-1
\right)/Z
}_{\text{mass shift}}
\times
\underbrace{
\left(
\mathbf v_t(\mathbf x_t|\mathbf z)
-
\mathbf v^{\mathrm{o}}(\mathbf x_t,t)
\right)
}_{\text{velocity discrepancy}}.
\end{align}

\noindent\textbf{Objective.}
We train the parameterized velocity field $\mathbf v_{\boldsymbol{\theta}}$
to approximate $\mathbf v^n$ by minimizing the following flow matching
objective:
\begin{align}
\mathcal{L}_{\mathrm{RCFM}}
&=
\mathbb E_t
\mathbb E_{
\mathbf z\sim\pi^n(\mathbf z),
\mathbf x_t\sim p_t(\cdot|\mathbf z)
}
\left[
\left\|
\mathbf v_{\boldsymbol{\theta}}(\mathbf x_t,t)
-

\right\|_2^2
\right]\\

\mathbb E_t
\mathbb E_{
\mathbf z\sim\pi^{\mathrm{o}}(\mathbf z),
\mathbf x_t\sim p_t(\cdot|\mathbf z)

{\pi^{\mathrm{o}}(\mathbf z)}

\mathbf v_{\boldsymbol{\theta}}(\mathbf x_t,t)
-

\right\|_2^2
\right].
\end{align}

\subsection{Implementation}
\noindent\textbf{Monte Carlo form.}
The reward-tilted velocity field can be viewed as a correction of the
original velocity field $\mathbf v_{\mathrm{old}}(\mathbf x_t,t)$.
Instead of explicitly modifying the sampling dynamics, we learn a new
velocity estimator by matching the model prediction to the reward-induced
target velocity. Specifically, the correction term is 
\begin{align}
\mathbf v^n(\mathbf x_t,t)
=
\mathbf v^{\mathrm{o}}(\mathbf x_t,t)
+ 
\mathbf c_t(\mathbf x_t,\mathbf z),
\end{align}
where $\mathbf c_t(\mathbf x_t,\mathbf z)$ represents the reward-induced
velocity correction and $\beta/\bar Z$ controls its magnitude. Therefore,
the updated velocity field can be obtained by minimizing the following
regression objective:
\begin{align}
\min_{\theta}
\mathbb{E}_{t, \mathbf z\sim\pi^{\mathrm{o}}(\mathbf z),\mathbf x_t\sim p_t(\cdot|\mathbf z)}
\left[
w_{\mathcal D}(\mathbf z)
\left\|
\mathbf v_{\boldsymbol{\theta}}(\mathbf x_t,t)
-
\mathtt{sg}(
\mathbf v^{\mathrm{o}}(\mathbf x_t,t)
+ 
\mathbf c_t(\mathbf x_t,\mathbf z))
\right\|_2^2
\right].
\end{align}

This objective encourages the new velocity field to preserve the original
generative dynamics while incorporating the reward-induced correction.
Compared with direct reward-guided sampling, the reward signal is absorbed
into the learned vector field, enabling efficient sampling without
evaluating the reward model during inference.


\noindent\textbf{On-policy or one-step gradient descent.}
The gradient is as follows,
\begin{align}
~&-2 w_{\mathcal D}(\mathbf z)
\frac{\partial\mathbf{v}_{\boldsymbol{\theta}}(\mathbf x_t,t)}
{\partial \boldsymbol{\theta}}
\mathbf c_t(\mathbf x_t,\mathbf z)) \\
 =~& 
 -2\underbrace{
 \frac{\pi^n(\mathbf z)}
 {\pi^{\mathrm{o}}(\mathbf z)}
 }_{\text{importance weight}}
 \times
 \underbrace{
 \alpha^o_t(\mathbf x_t,\mathbf z)
 }_{\text{trajectory contribution}}
 \times
 \underbrace{
 \frac{\beta
 \left(
 \exp(r(\mathbf z)/\gamma)-1
 \right)}
 {Z}
 }_{\text{mass shift}}
 \times
 \underbrace{\frac{\partial\mathbf{v}_{\boldsymbol{\theta}}(\mathbf x_t,t)}
 {\partial \boldsymbol{\theta}}}_{\text{Jacobian}}
 \times
 \underbrace{
 \left(
 \mathbf v_t(\mathbf x_t|\mathbf z)
 -
 \mathbf v_{\boldsymbol{\theta}}(\mathbf x_t,t)
 \right)
 }_{\text{velocity discrepancy}}.
\end{align}

The last two terms on the right-hand side correspond to the standard flow matching
gradient, where the velocity discrepancy provides the regression direction and the
Jacobian term propagates this discrepancy to the model parameters.


The proposed formulation equivalently introduces an adaptive weighting mechanism on top of the
standard flow matching objective.
The overall update can be viewed as an adaptive flow matching update,
where the effective learning signal is modulated by the adaptive factor
\begin{align}
w_{\mathrm{adapt}}(\mathbf x_t,\mathbf z)
=
\frac{\pi^n(\mathbf z)}
{\pi^{\mathrm{o}}(\mathbf z)}
\alpha_t^o(\mathbf x_t,\mathbf z)
\frac{\beta
\left(
\exp(r(\mathbf z)/\gamma)-1
\right)}
{Z}.
\end{align}
Compared with conventional flow matching, which treats all training pairs equally,
the proposed weighting dynamically emphasizes samples that are more aligned with
the optimal policy, have higher contribution to the transport trajectory, and
receive higher reward.



\noindent\textbf{\color{red}Trajectory contribution weight.}
Although the exact trajectory contribution weight
\begin{align}
\alpha_t(\mathbf{x}_t,\mathbf z)
=
\frac{
p_t(\mathbf{x}_t|\mathbf z)
}
{
p_t(\mathbf{x}_t)
}
\end{align}
is theoretically well-defined, directly estimating this density ratio is
challenging since the marginal
density may be intractable. 

Instead, we exploit the posterior averaging property of the optimal velocity
field:
\begin{align}
\mathbf v(\mathbf{x}_t,t)
=
\mathbb{E}_{\mathbf z\sim\pi}
\left[
\alpha_t(\mathbf{x}_t,\mathbf z)
\mathbf v_t(\mathbf{x}_t|\mathbf z)
\right].
\end{align}
A trajectory
with a larger posterior contribution
$\pi(\mathbf{z})\alpha_t(\mathbf{x}_t,\mathbf z)$ is expected to have a conditional velocity
that is more consistent with the optimal velocity field
$\mathbf v(\mathbf{x}_t,t)$. Therefore, for sample $i$ in the current training
group $\mathcal G$, we measure its velocity discrepancy by the mean absolute
error over all non-batch dimensions:
\begin{align}
d_{t,i}
=
\operatorname{mean}_{\mathrm{feat}}
\left|
\mathbf v_i-
\mathbf v_i^{\mathrm{old}}
\right|,
\end{align}
where $\mathbf v_i$ is the conditional target velocity and
$\mathbf v_i^{\mathrm{old}}$ is the old-model velocity prediction. We first form
the unnormalized reciprocal-discrepancy score
\begin{align}
\tilde{\alpha}_{t,i}
=
\frac{1}{d_{t,i}+\epsilon_{\mathrm{alg}}}.
\end{align}
We then normalize these scores within the current training group:
\begin{align}
\alpha_{t,i}^{\mathcal G}
&=
\frac{
\tilde{\alpha}_{t,i}
}{
|\mathcal G|^{-1}
\sum_{j\in\mathcal G}\tilde{\alpha}_{t,j}
},
&
\frac{1}{|\mathcal G|}
\sum_{i\in\mathcal G}\alpha_{t,i}^{\mathcal G}
&=1.
\end{align}
In the implementation, $\mathcal G$ is the current per-device training
minibatch at a fixed training timestep. Both the reciprocal score and its
group mean are computed with stopped gradients. This empirical unit-mean
constraint matches the theoretical identity
$\mathbb{E}_{\mathbf z}[\alpha_t(\mathbf{x}_t,\mathbf z)]=1$, while preserving
the relative trajectory contributions inside each group and preventing their
overall scale from changing the correction magnitude across minibatches.

{\color{red}
The intuition is that the velocity discrepancy is jointly determined by the
trajectory contribution factor $\alpha_t(\mathbf{x}_t,\mathbf z)$ and the source
probability $\pi(\mathbf z)$. Since $\mathbf z$ is sampled from $\pi(\mathbf z)$,
the sampled trajectories typically have non-negligible probability mass.
Therefore, variations in the velocity discrepancy mainly reflect the relative
contribution factor $\alpha_t(\mathbf{x}_t,\mathbf z)$. In particular, a small
velocity discrepancy indicates that the trajectory is more consistent with the
optimal velocity field and is likely associated with a larger contribution
factor, whereas a large velocity discrepancy suggests a smaller contribution
factor. The group normalization retains this relative ordering but calibrates
the average factor to one within every training group.
}

{\color{red}
Note that this does not provide an exact estimation of

\noindent\textbf{Importance weights.}
We need to compute unnormalized importance weight, $\bar{w}(\mathbf{z}) = \exp(r(\mathbf z)/\gamma)$, 
for reward-guided mass shift.

\emph{GRPO-style:} 
At each training iteration, let $R_{j,k}$ be the reward of rollout $k$
for prompt $c_j$, where $j=1,\ldots,C$, $k=1,\ldots,K$, and
$D=CK$. We center rewards by the mean of their prompt group,
\begin{align}
\bar R_{c_j}
=
\frac{1}{K}\sum_{k=1}^{K}R_{j,k},
\end{align}
and use the population standard deviation over all rollout rewards in the
current iteration as the common reward scale:
\begin{align}
\bar R
&=
\frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K}R_{j,k},\\
\sigma_R
&=
\sqrt{
\frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K}
(R_{j,k}-\bar R)^2
}
+\epsilon_R.
\end{align}
Given the mass-shift clip $\Delta_{\mathrm{clip}}>0$, the signed mass shift
and unnormalized importance weight are
\begin{align}
\Delta_{j,k}
&=
\frac{1}{\Delta_{\mathrm{clip}}}
\operatorname{clip}
\left(
\frac{R_{j,k}-\bar R_{c_j}}{\sigma_R},
-\Delta_{\mathrm{clip}},
\Delta_{\mathrm{clip}}
\right),\\
W_{j,k}&=1+\Delta_{j,k}.
\end{align}
Thus, rewards are centered within each prompt group but share a global scale
computed from the current rollout iteration. For asymmetric mass shifts, we
use $\sigma_+=c\sigma_R$ and $\sigma_-=\sigma_R/c$, as detailed in the
ablation study.



One similar choice is derived 
by miximizing
\begin{align}
    \sum_i r_i w_i
\end{align}
with the constraint $\sum_i w^2_i = c$:
then $w_i = \frac{r_i}{\frac{1}{N}\sum_{j} r_j}
= \frac{r_i}{\beta + \beta/D \sum_l r_j}$. 
$r_j$ (regarded as $\bar{w}(\mathbf{z}) = \exp(r(\mathbf{z})/\gamma)$) is normalized so that
the reward of background sample is $1$,
larger than $1$ is good,
smaller than $1$ is bad,
of course it could also be clipped.

\emph{$\mathtt{Softmax}$-style:}
$r$ is adjusted so that $r=0$
for the background sample.
\begin{align}
    \bar{w}(\mathbf{z}) = \frac{\exp(r/t))}
    {\beta + \beta/D\sum_{i=1}^D\exp(r_i/t)}
\end{align}
$t$ plays a role,
something like the $\mathtt{clip}$ operator,
transforming $r$ to a probability.
If $t=0$, it becomes $\mathtt{max}$
If $t=\infty$,
it becomes uniform, all are $1$.
\\

\subsection{Experimental setup}
\label{sec: exp-setting}
We follow the experimental setup used by DiffusionNFT~\citep{zheng2026diffusionnft}. Unless otherwise specified, all experiments start from SD3.5-Medium~\citep{esser2024scaling} at $512 \times 512$ resolution.

\paragraph{Reward models.}
We evaluate both rule-based and model-based reward optimization. The rule-based tasks are GenEval for compositional text-to-image alignment~\citep{ghosh2023geneval} and OCR visual text rendering, using the same prompt splits and partial-credit reward design as the DiffusionNFT and FlowGRPO. The model-based rewards include PickScore~\citep{kirstain2023pickapic}, CLIPScore~\citep{hessel2021clipscore}, HPSv2.1~\citep{wu2023human}, Aesthetic score~\citep{schuhmann2022laion}, ImageReward~\citep{xu2023imagereward}, and UnifiedReward~\citep{wang2025unifiedreward}. Following DiffusionNFT, Pick-a-Pic prompts are used for preference-oriented training~\citep{kirstain2023pickapic}, while DrawBench prompts are reserved for out-of-domain evaluation~\citep{saharia2022photorealistic}.

\paragraph{Training protocol.}
We fine-tune the base model with LoRA~\citep{hu2022lora} using rank $r=32$, scaling factor $\alpha=64$, and learning rate $3 \times 10^{-4}$. Each training epoch contains 48 prompt groups, and each group samples $G=24$ images from the current rollout policy. Before constructing the mass shifts, rewards are centered by the mean of their prompt group and scaled by the population standard deviation over all rollout rewards in the current training iteration. For single-reward comparisons and ablations, rollouts use 10 sampling steps to match the efficient setting in DiffusionNFT. For multi-reward training and final evaluation, we use 40 sampling steps to reduce sampler-induced quality differences.

\subsubsection{Ablation on Forward Timestep Sampling}

We investigate the effect of different forward-timestep sampling strategies during training.
Across all settings, the sampled values are transformed using the same timestep-shift function:
\begin{equation}
\operatorname{shift}(x)
=
\frac{3x}{1+2x}.
\label{eq:timestep_shift}
\end{equation}

We compare the following three sampling schemes.

\medskip
\noindent\textbf{1. DiffusionNFT Original.}
Following the original DiffusionNFT implementation, we construct 10 uniformly spaced candidate points in $[0.001,1]$, apply the shift transformation, and uniformly sample 9 distinct timesteps without replacement:
\begin{align*}
x_j &= \operatorname{LinearSpace}(0.001,1,10)_j,
\qquad
T_j = \operatorname{shift}(x_j),
\quad j=0,\ldots,9,\\
\{t_0,\ldots,t_8\}
&\sim
\operatorname{SampleWithoutReplacement}
\left(
\{T_0,\ldots,T_9\},\,9
\right).
\end{align*}
This setting corresponds to the \textbf{green curve}.

\medskip
\noindent\textbf{2. Logit-Normal Sampling.}
Instead of sampling from a fixed discrete candidate set, we independently draw each timestep from a logit-normal distribution and then apply the same shift transformation:
\begin{align*}
z_i &\overset{\mathrm{iid}}{\sim} \mathcal N(0,1),
\qquad

The original training objective is

$$
\mathcal{L}
=
\left(
\Delta_{\mathrm{clip}}\mathcal{L}_{\mathrm{target}}
+
\lambda_{\mathrm{ref}}\mathcal{L}_{\mathrm{ref}}
\right),
$$

where $\Delta_{\mathrm{clip}}=5$ and $\lambda_{\mathrm{ref}}=10^{-4}$. Since training uses AdamW, scaling the entire loss by a positive constant approximately preserves the parameter update. Therefore, dividing the objective by $\Delta_{\mathrm{clip}}$ gives the approximately equivalent formulation

$$
\mathcal{L}
=\mathcal{L}_{\mathrm{target}}
+
\frac{\lambda_{\mathrm{ref}}}{\Delta_{\mathrm{clip}}}\mathcal{L}_{\mathrm{ref}},
$$


We further evaluate a direct removal of $\Delta_{\mathrm{clip}}$ without rescaling $\lambda_{\mathrm{ref}}$:

$$
\mathcal{L}
=
\mathcal{L}_{\mathrm{target}}
+
\lambda_{\mathrm{ref}}\mathcal{L}_{\mathrm{ref}}.
$$

We compare these formulations under the single-reward GenEval setting to examine whether the explicit mass-shift-clip scaling is necessary and how the resulting change in target--reference balance affects training.

\subsubsection{Asymmetric Scaling for Positive and Negative Mass Shifts}

For the $D=CK$ rollout samples in the current training iteration, let the
global reward mean and population standard deviation be

\begin{align}
\bar R
&=
\frac{1}{D}
\sum_{j=1}^{C}\sum_{k=1}^{K}R_{j,k},\\
\sigma_R
&=
\sqrt{
\frac{1}{D}
\sum_{j=1}^{C}\sum_{k=1}^{K}
(R_{j,k}-\bar R)^2
}
+\epsilon_R.
\end{align}
The same $\sigma_R$ is shared by every prompt group in the iteration, while
each reward is centered by its prompt-group mean
$\bar R_{c_j}=K^{-1}\sum_{k=1}^{K}R_{j,k}$.

To isolate the effect of asymmetric scaling between positive and negative mass shifts, we apply a fixed scaling factor $c>0$ to the normalization scale:

\begin{align}
\sigma_+ = c\sigma_R, \quad
\sigma_- = \frac{\sigma_R}{c}.
\end{align}

For a sample $i$ conditioned on prompt $c_i$, the resulting signed mass shift
is

\begin{equation}
\Delta_i^{(c)}=
\begin{cases}
\displaystyle
\frac{1}{\Delta_{\mathrm{clip}}}
\operatorname{clip}
\left(
\frac{R_i-\bar R_{c_i}}{c\sigma_R},
0,
\Delta_{\mathrm{clip}}
\right),
& R_i-\bar R_{c_i}>0,\\[10pt]
\displaystyle
\frac{1}{\Delta_{\mathrm{clip}}}
\operatorname{clip}
\left(
\frac{R_i-\bar R_{c_i}}{\sigma_R/c},
-\Delta_{\mathrm{clip}},
0
\right),
& R_i-\bar R_{c_i}\le 0.
\end{cases}
\end{equation}

We consider the following candidate values for the scaling factor:

\begin{equation}
c \in
\left{
\frac{1}{2},
\frac{1}{\sqrt{2}},
1,
\sqrt{2},
2
\right}.
\end{equation}

Here, $c=1$ exactly recovers the symmetric mass-shift construction under the
same prompt centering and global reward scale. Values $c>1$ emphasize negative
mass shifts, whereas values $c<1$ emphasize positive mass shifts.

\subsection{A Unified View of Diffusion RL Objectives}
\label{sec:velocity_target_objectives}

We revisit existing RL objectives for flow matching models from a unified
velocity target perspective. Although these methods are motivated by different
RL principles, we show that they can all be interpreted as specifying: (i) an
effective velocity target for each generated sample, and (ii) a reward- or
advantage-dependent weight that determines how strongly the current model should
match or avoid this target.

We focus on one generated sample $i \in {1, \cdots, G}$ at timestep $t$.
% and omit timestep
% averaging and implementation details that are not essential to this view. 
Let
\[
\mathbf{v}_i^\theta
=
\mathbf{v}_\theta(\mathbf{x}_{i}^t,t,c_i),
\qquad
\mathbf{v}_i^{\mathrm{old}}
=
\mathbf{v}_{\theta_{\mathrm{old}}}(\mathbf{x}_{i}^t,t,c_i),
\]
denote the current and old velocity, respectively. Let
$\mathbf{v}_i$ denote the sample $i$ velocity, i.e., 
\begin{equation}
\mathbf{x}_{i}^t
=
(1-t)\mathbf{x}_i^t
+
t\boldsymbol{\epsilon}_i,
\qquad
\mathbf{v}_i
=
\boldsymbol{\epsilon}_i-\mathbf{x}_i^0 .
\end{equation}

Under this notation, a broad class of diffusion/flow RL objectives can be
viewed through the following  form:
\begin{equation}
\label{eq:unified_form}
\ell_i(\theta)
\approx
\kappa_i(A_i,t)
\left\|
\mathbf{v}_i^\theta
-
\mathbf{\tau}_i
\left(
A_i,
\mathbf{v}_i,
\mathbf{v}_i^{\mathrm{old}},
t
\right)
\right\|^2
+
\mathcal{R}_i(\theta),
\end{equation}
where $\mathbf{\tau}_i(\cdot)$ is the pseudo target function, it receives advantage $A_i$, currrent, old velocity $\mathbf{v}_i^{\theta}, \mathbf{v}_i^{old}$, and timestep $t$ as inputs.


$\kappa_i(A_i,t)$ is the weight, receives advantage $A_i$ and timestep $t$ as inputs. Intuitively, it
determines how strongly the current model should match or
avoid this target.

% and $\mathbf{v}_i^{\mathrm{ref}}$ denotes the velocity
% prediction of a frozen reference model.

The second term $\mathcal{R}_i(\theta)$  is the 
regularization term, commonly used to prevent the fine-tuned model
from drifting too far from the reference velocity field. The term
denotes additional proximal constraints, such as the
GRPO-style clipping of the likelihood ratio in Flow-GRPO. DiffusionNFT and
AWM use MSE regularization with reference model prediction.



\subsubsection{Flow-GRPO}
\label{sec:flow_grpo}

Flow-GRPO starts from an policy-gradient objective over stochastic
denoising transitions. For one transition from timestep $t$ to $s$, the
sample-level objective (the following derivation corresponds to the first term in Eq.~\ref{eq:unified_form}) can be written as
\begin{equation}
\ell_i^{\mathrm{FG}}(\theta)
=
-
A_i
\rho_i(\theta),
\qquad
\rho_i(\theta)
=
\frac{
p_{\theta}(\mathbf{x}_i^s\mid \mathbf{x}_i^t,c_i)
}{
p_{\theta_{\mathrm{old}}}(\mathbf{x}_i^s\mid \mathbf{x}_i^t,c_i)
},
\end{equation}
where $A_i$ is the trajectory-level advantage. Assume that the reverse
transition is modeled as a Gaussian policy with shared covariance:
\begin{equation}
p_{\theta}(\mathbf{x}^{s}_{i}\mid \mathbf{x}^{t}_{i},c_i)
=
\mathcal{N}
\left(
\mathbf{x}^{s}_{i};
a_t\mathbf{x}^{t}_{i}+b_t\mathbf{v}_i^\theta,
\eta_t^2 I
\right)
\end{equation}
For a fixed sampled transition $(\mathbf{x}_i^t,\mathbf{x}_i^s)$, this Gaussian
mean induces a pseudo target:
\begin{equation}
\label{eq:flow_grpo_pseudo_target_short}
\mathbf{\tau}_i^{\mathrm{FG}}
=
\frac{\mathbf{x}^{s}_{i}-a_t\mathbf{x}^{t}_{i}}{b_t}
\end{equation}
Since
\begin{equation}
\mathbf{x}^{s}_{i}
=
a_t\mathbf{x}^{t}_{i}
+
b_t\mathbf{v}_i^{\mathrm{old}}
+
\eta_t\boldsymbol{\epsilon}_i,
\end{equation}
the pseudo target can also be written as
\begin{equation}
\label{eq:flow_grpo_target_short}
\mathbf{\tau}_i^{\mathrm{FG}}
=
\mathbf{v}_i^{\mathrm{old}}
+
\frac{\eta_t}{b_t}
\boldsymbol{\epsilon}_i .
\end{equation}
Thus, the Flow-GRPO target is a stochastic noise add velocity target induced by the sampled old-policy
transition.

Substituting Eq.~\ref{eq:flow_grpo_pseudo_target_short} into the Gaussian
likelihood gives
\begin{equation}
\rho_i(\theta)
=
C_i
\exp
\left[
-
\frac{b_t^2}{2\eta_t^2}
\left\|
\mathbf{v}_i^\theta
-
\mathbf{\tau}_i^{\mathrm{FG}}
\right\|^2
\right],
\end{equation}
where
\begin{equation}
C_i
=
\exp
\left[
\frac{b_t^2}{2\eta_t^2}
\left\|
\mathbf{v}_i^{\mathrm{old}}
-
\mathbf{\tau}_i^{\mathrm{FG}}
\right\|^2
\right]
\end{equation}
is independent of $\theta$. Therefore, the Flow-GRPO objective can be
viewed as an advantage-weighted function of the squared distance between
$\mathbf{v}_i^\theta$ and the transition-induced target
$\mathbf{\tau}_i^{\mathrm{FG}}$.

To connect it with the unified target-matching form, consider the gradient of
the sample objective with respect to the velocity prediction:
\begin{equation}
\nabla_{\mathbf{v}_i^\theta}
\ell_i^{\mathrm{FG}}(\theta)
=
2
\left(
A_i
\frac{b_t^2}{2\eta_t^2}
\rho_i(\theta)
\right)
\left(
\mathbf{v}_i^\theta
-
\mathbf{\tau}_i^{\mathrm{FG}}
\right).
\end{equation}
This has the same local gradient direction as a weighted squared-error objective
\begin{equation}
\kappa_i
\left\|
\mathbf{v}_i^\theta
-
\mathbf{\tau}_i^{\mathrm{FG}}
\right\|^2.
\end{equation}


In this sense, Flow-GRPO fits the unified form in
Eq.~\ref{eq:unified_form} with
\begin{equation}
\mathbf{\tau}_i
=
\mathbf{v}_i^{\mathrm{old}}
+
\frac{\eta_t}{b_t}
\boldsymbol{\epsilon}_i ,
\qquad
\kappa_i
=
A_i
\frac{b_t^2}{2\eta_t^2} \rho_i(\theta).
\end{equation}
The sign of $A_i$ determines whether the model is pulled toward or pushed away
from the pseudo target. A positive advantage makes gradient descent move
$\mathbf{v}_i^\theta$ toward the sampled transition-level target, whereas a
negative advantage reverses this direction. The main distinction from clean
flow-matching objectives is that this target contains transition noise through
$\frac{\eta_t}{b_t}\boldsymbol{\epsilon}_i$, rather than being defined by the
clean residual $\mathbf{v}_i-\mathbf{v}_i^{\mathrm{old}}$.



\subsubsection{DiffusionNFT}
\label{sec:diffusionnft}

DiffusionNFT constructs positive and negative velocity branches around the old
velocity prediction:
\begin{equation}
\mathbf{v}_{i}^{+}
=
\mathbf{v}_i^{\mathrm{old}}
+
\beta
\left(
\mathbf{v}_i^\theta-\mathbf{v}_i^{\mathrm{old}}
\right),
\qquad
\mathbf{v}_{i}^{-}
=
\mathbf{v}_i^{\mathrm{old}}
-
\beta
\left(
\mathbf{v}_i^\theta-\mathbf{v}_i^{\mathrm{old}}
\right).
\end{equation}
Given an optimality reward $r_i\in[0,1]$, its sample-level loss is
\begin{equation}
\label{eq:nft_velocity_target_origin}
\ell_i^{\mathrm{NFT}}(\theta)
=
r_i
\left\|
\mathbf{v}_{i}^{+}
-
\mathbf{v}_i
\right\|^2
+
(1-r_i)
\left\|
\mathbf{v}_{i}^{-}
-
\mathbf{v}_i
\right\|^2 .
\end{equation}

This loss is equivalent, up to constants independent of $\theta$, to
\begin{equation}
\label{eq:nft_velocity_target_short}
\ell_i^{\mathrm{NFT}}(\theta)
=
\beta^2
\left\|
\mathbf{v}_i^\theta
-
\mathbf{\tau}_i^{\mathrm{NFT}}
\right\|^2,
\end{equation}
where
\begin{equation}
\label{eq:nft_pseudo_target_short}
\mathbf{\tau}_i^{\mathrm{NFT}}
=
\mathbf{v}_i^{\mathrm{old}}
+
\frac{2r_i-1}{\beta}
\left(
\mathbf{v}_i-\mathbf{v}_i^{\mathrm{old}}
\right).
\end{equation}

DiffusionNFT therefore fits Eq.~\ref{eq:unified_form}, with
\begin{equation}
\mathbf{\tau}_i
=
\mathbf{v}_i^{\mathrm{old}}
+
\frac{2r_i-1}{\beta}
\left(
\mathbf{v}_i-\mathbf{v}_i^{\mathrm{old}}
\right),
\qquad
\kappa_i
=
\beta^2 .
\end{equation}
Unlike Flow-GRPO, its reward directly determines where the effective target is placed along
the clean residual direction
\[
\mathbf{v}_i-\mathbf{v}_i^{\mathrm{old}}.
\]
When $r_i>0.5$, the target moves from the old velocity toward the clean
flow-matching target; when $r_i<0.5$, it is placed on the opposite side of the
old velocity. Thus, DiffusionNFT can be viewed as reward-conditioned velocity
target displacement.

The same interpretation can be expressed in the corresponding
$\mathbf{x}_0$-prediction space. Under the parameterization
\begin{equation}
\mathbf{x}_{0,i}^{\theta}
=
\mathbf{x}_{t,i}
-
t\mathbf{v}_i^\theta,
\qquad
\mathbf{x}_{0,i}^{\mathrm{old}}
=
\mathbf{x}_{t,i}
-
t\mathbf{v}_i^{\mathrm{old}},
\qquad
\mathbf{x}_{0,i}
=
\mathbf{x}_{t,i}
-
t\mathbf{v}_i,
\end{equation}
define the induced pseudo-target
\begin{equation}
x_{0,i}^{\mathrm{NFT}}
=
\mathbf{x}_{t,i}
-
t\mathbf{\tau}_i^{\mathrm{NFT}}.
\end{equation}
Substituting Eq.~\ref{eq:nft_pseudo_target_short} gives
\begin{align}
x_{0,i}^{\mathrm{NFT}}
&=
\mathbf{x}_{0,i}^{\mathrm{old}}
-
\frac{t(2r_i-1)}{\beta}
\left(
\mathbf{v}_i-\mathbf{v}_i^{\mathrm{old}}
\right)
\\
&=
\mathbf{x}_{0,i}^{\mathrm{old}}
+
\frac{2r_i-1}{\beta}
\left(
\mathbf{x}_{0,i}-\mathbf{x}_{0,i}^{\mathrm{old}}
\right).
\end{align}
Moreover,
\begin{equation}
\mathbf{v}_i^\theta-\mathbf{\tau}_i^{\mathrm{NFT}}
=
-\frac{1}{t}
\left(
\mathbf{x}_{0,i}^\theta
-
x_{0,i}^{\mathrm{NFT}}
\right),
\end{equation}
and in the implementation, $\ell_i^{\mathrm{NFT}}(\theta) = \frac{1}{\beta} \ell_i^{\mathrm{NFT}}(\theta)$,

hence the velocity-prediction losses in Eq.~\ref{eq:nft_velocity_target_origin} and
Eq.~\ref{eq:nft_velocity_target_short} are equivalently


\begin{equation}
\ell_i^{\mathrm{NFT}}(\theta)
=
\frac{1}{\beta t^2}(r_i
\left\|
\mathbf{x}_{i}^{+}
-
\mathbf{x}_i
\right\|^2
+
(1-r_i)
\left\|
\mathbf{x}_{i}^{-}
-
\mathbf{x}_i
\right\|^2) .
\end{equation}


\begin{equation}
\ell_i^{\mathrm{NFT}}(\theta)
=
\frac{\beta}{t^2}
\left\|
\mathbf{x}_{0,i}^{\theta}
-
x_{0,i}^{\mathrm{NFT}}
\right\|^2.
\end{equation}

\begin{equation}
x_{0,i}^{\mathrm{NFT}}
=
\mathbf{x}_{0,i}^{\mathrm{old}}
+
\frac{2r_i-1}{\beta}
\left(
\mathbf{x}_{0,i}-\mathbf{x}_{0,i}^{\mathrm{old}}
\right).
\end{equation}




\paragraph{Adaptive loss weighting in the implementation.}
The practical implementation of DiffusionNFT applies adaptive loss weighting
separately to the positive and negative branches. For a tensor
$\mathbf{u}\in\mathbb{R}^{D}$, define
\begin{equation}
\|\mathbf{u}\|^2
=
\frac{1}{D}\sum_{j=1}^{D}u_j^2,
\qquad
m(\mathbf{u})
=
\frac{1}{D}\sum_{j=1}^{D}|u_j|.
\end{equation}
The two branch predictions in $\mathbf{x}_0$-space are
\begin{align}
\mathbf{x}_{0,i}^{+}
&=
\mathbf{x}_{0,i}^{\mathrm{old}}
+
\beta
\left(
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{old}}
\right),
\\
\mathbf{x}_{0,i}^{-}
&=
\mathbf{x}_{0,i}^{\mathrm{old}}
-
\beta
\left(
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{old}}
\right).
\end{align}
The corresponding detached adaptive normalization factors are
\begin{align}
w_i^{+}(t)
&=
\operatorname{sg}
\left[
\max
\left(
m\left(
\mathbf{x}_{0,i}^{+}
-
\mathbf{x}_{0,i}
\right),
\epsilon
\right)
\right],
\\
w_i^{-}(t)
&=
\operatorname{sg}
\left[
\max
\left(
m\left(
\mathbf{x}_{0,i}^{-}
-
\mathbf{x}_{0,i}
\right),
\epsilon
\right)
\right].
\label{eq:origin-weight-factor}
\end{align}
Up to an additional global advantage scaling factor, the implementation uses
the sample-level objective
\begin{align}
\label{eq:nft_adaptive_separate}
\ell_{i,\mathrm{sep}}^{\mathrm{NFT\text{-}AW}}
=
\frac{1}{\beta}
\Bigg[
&
\frac{r_i}{w_i^{+}(t)}
\left\|
\mathbf{x}_{0,i}^{+}
-
\mathbf{x}_{0,i}
\right\|^2
+
\frac{1-r_i}{w_i^{-}(t)}
\left\|
\mathbf{x}_{0,i}^{-}
-
\mathbf{x}_{0,i}
\right\|^2
\Bigg].
\end{align}
The common factor $1/\beta$ follows the implementation and affects only the
overall loss scale.



\paragraph{Exact single-branch form.}Since
$w_i^{+}(t)$ and $w_i^{-}(t)$ are detached, completing the square in
Eq.~\ref{eq:nft_adaptive_separate} gives the following equality of gradients:
\begin{align}
\label{eq:nft_adaptive_single_gradient}
\nabla_\theta
\ell_{i,\mathrm{sep}}^{\mathrm{NFT\text{-}AW}}
=
\nabla_\theta
\Bigg\{
&\beta
\left(
\frac{r_i}{w_i^{+}(t)}
+
\frac{1-r_i}{w_i^{-}(t)}
\right)
\Bigg\|
\mathbf{x}_{0,i}^{\theta}
\nonumber\\
&\quad-
\left[
\mathbf{x}_{0,i}^{\mathrm{old}}
+
\frac{
\displaystyle
\frac{r_i}{w_i^{+}(t)}
-
\frac{1-r_i}{w_i^{-}(t)}
}{
\displaystyle
\beta
\left(
\frac{r_i}{w_i^{+}(t)}
+
\frac{1-r_i}{w_i^{-}(t)}
\right)
}
\left(
\mathbf{x}_{0,i}
-
\mathbf{x}_{0,i}^{\mathrm{old}}
\right) \right]
\Bigg\|^2
\Bigg\}.
\end{align}
The omitted difference between the two losses is independent of $\theta$ and
therefore has no effect on training. This expression shows the two effects of
the original adaptive factors directly. Their weighted inverse sum controls
the overall gradient scale, while their weighted inverse difference changes
the pseudo-target. 


\paragraph{Direct post-merge adaptive weighting.}
A much simpler choice is to apply the same normalization directly to the
single-branch residual:
\begin{equation}
\label{eq:nft_postmerge_adaptive}
\ell_{i,\mathrm{post}}^{\mathrm{NFT\text{-}AW}}
=
\beta
\frac{
\left\|
\mathbf{x}_{0,i}^{\theta}
-
x_{0,i}^{\mathrm{NFT}}
\right\|^2
}{
\operatorname{sg}
\left[
\max
\left(
m\left(
\mathbf{x}_{0,i}^{\theta}
-
x_{0,i}^{\mathrm{NFT}}
\right),
\epsilon
\right)
\right]
}.
\end{equation}
The leading $\beta$ matches the current single-branch implementation. This
objective is not strictly equivalent to Eq.~\ref{eq:nft_adaptive_separate},
because it uses one residual and cannot reproduce the target change caused by
unequal $w_i^{+}(t)$ and $w_i^{-}(t)$. Its denominator is the
error of the prediction that is actually optimized. 






\paragraph{Branch-free adaptive weighting.}
The adaptive weight should measure the prediction difficulty of sample \(i\) at
timestep \(t\). In the original NFT implementation, this difficulty is
estimated from the two branch predictions in
Eq.~\ref{eq:origin-weight-factor}. Since both branch predictions are
independent of the reward, their reconstruction errors relative to the clean
target provide stable estimates of sample and timestep difficulty. The reward
only determines how the two branch errors are combined.

This property is weakened by direct post-merge adaptive weighting. Let
\begin{equation}
A_i=2r_i-1,
\end{equation}
and define the single NFT pseudo-target as
\begin{equation}
\mathbf{x}_{0,i}^{\mathrm{NFT}}
=
\mathbf{x}_{0,i}^{\mathrm{old}}
-
\frac{A_i}{\beta}
\left(
\mathbf{x}_{0,i}^{\mathrm{old}}
-
\mathbf{x}_{0,i}
\right).
\end{equation}
Its fitting residual is
\begin{equation}
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{NFT}}
=
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{old}}
+
\frac{A_i}{\beta}
\left(
\mathbf{x}_{0,i}^{\mathrm{old}}
-
\mathbf{x}_{0,i}
\right).
\label{eq:nft_single_target_residual}
\end{equation}
Using this residual directly as the adaptive scale makes the estimated
difficulty strongly reward-dependent. When \(A_i\approx0\), or equivalently
\(r_i\approx1/2\), the pseudo-target approaches the old prediction and the
residual becomes dominated by
\(\mathbf{x}_{0,i}^{\theta}-\mathbf{x}_{0,i}^{\mathrm{old}}\). It may therefore
be small simply because the current policy remains close to the old policy,
even when the old reconstruction error
\(\mathbf{x}_{0,i}^{\mathrm{old}}-\mathbf{x}_{0,i}\) is large. In this case,
the adaptive factor no longer measures the actual difficulty of the sample and
timestep.

The factor \(1/\beta\) in
Eq.~\ref{eq:nft_single_target_residual} also introduces an artificial scale
dependence. We first remove it by multiplying the residual by \(\beta\):
\begin{equation}
\beta
\left(
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{NFT}}
\right)
=
\beta
\left(
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{old}}
\right)
+
A_i
\left(
\mathbf{x}_{0,i}^{\mathrm{old}}
-
\mathbf{x}_{0,i}
\right).
\end{equation}
However, rescaling alone is insufficient: when \(A_i\approx0\), the old
reconstruction error still disappears from this residual. We therefore add an
old-error anchor whose contribution increases as the advantage approaches
zero. For each tensor element \(j\), define
\begin{align}
q_{i,j}(r_i,t)
&=
\beta^2
\left(
x_{0,i,j}^{\theta}
-
x_{0,i,j}^{\mathrm{NFT}}
\right)^2
+
\left(1-A_i^2\right)
\left(
x_{0,i,j}^{\mathrm{old}}
-
x_{0,i,j}
\right)^2
\nonumber\\
&=
\Big[
\beta
\left(
x_{0,i,j}^{\theta}
-
x_{0,i,j}^{\mathrm{old}}
\right)
+
A_i
\left(
x_{0,i,j}^{\mathrm{old}}
-
x_{0,i,j}
\right)
\Big]^2
\nonumber\\
&\quad+
\left(1-A_i^2\right)
\left(
x_{0,i,j}^{\mathrm{old}}
-
x_{0,i,j}
\right)^2.
\label{eq:nft_branchfree_second_moment}
\end{align}
The first term measures the fitting error of the actual pseudo-target after
removing the artificial \(1/\beta\) scaling. The second term restores the old
reconstruction error when the reward provides little directional information.
Its coefficient satisfies
\begin{equation}
1-A_i^2
=
4r_i(1-r_i),
\end{equation}
which is largest at \(r_i=1/2\) and vanishes at the decisive endpoints
\(r_i=0\) and \(r_i=1\). Thus, when \(\lvert A_i\rvert\approx1\), the scale
relies mainly on the pseudo-target fitting residual. When \(A_i\approx0\), the
anchor preserves the underlying sample and timestep difficulty.

The resulting detached scale is
\begin{equation}
w_i(t)
=
\operatorname{sg}
\left[
\max
\left(
\frac{1}{D}
\sum_{j=1}^{D}
\sqrt{q_{i,j}(r_i,t)},
\epsilon
\right)
\right],
\label{eq:nft_branchfree_weight}
\end{equation}
and the objective becomes
\begin{equation}
\label{eq:nft_branchfree_adaptive}
\ell_i^{\mathrm{NFT\text{-}AW}}
=
\beta
\frac{
\left\|
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{NFT}}
\right\|^2
}{
w_i(t)
}.
\end{equation}

This construction retains a single pseudo-target and a single adaptive weight
while preserving a stable estimate of sample and timestep difficulty. At the
beginning of a policy update, when
\(\mathbf{x}_{0,i}^{\theta}=\mathbf{x}_{0,i}^{\mathrm{old}}\), we have
\begin{equation}
q_{i,j}
=
\left(
x_{0,i,j}^{\mathrm{old}}
-
x_{0,i,j}
\right)^2
\end{equation}
for every reward. At the neutral reward \(r_i=1/2\),
\begin{equation}
q_{i,j}
=
\left(
x_{0,i,j}^{\mathrm{old}}
-
x_{0,i,j}
\right)^2
+
\beta^2
\left(
x_{0,i,j}^{\theta}
-
x_{0,i,j}^{\mathrm{old}}
\right)^2,
\end{equation}
so the scale does not collapse merely because the current policy remains close
to the old policy.

Finally, the proposed quantity has a direct connection to the original two
branch errors:
\begin{equation}
q_{i,j}(r_i,t)
=
r_i
\left(
x_{0,i,j}^{+}
-
x_{0,i,j}
\right)^2
+
(1-r_i)
\left(
x_{0,i,j}^{-}
-
x_{0,i,j}
\right)^2.
\label{eq:nft_branchfree_branch_identity}
\end{equation}

Therefore, the proposed branch-free factor preserves the reward-weighted
second moment of the original two branch errors, while requiring only one
pseudo-target and one adaptive scale during optimization.


\paragraph{Simplified clean-target adaptive weighting.}
If the adaptive factor is y intended to measure the current difficulty of
sample \(i\) at timestep \(t\), an even simpler alternative is
\begin{equation}
\label{eq:nft_clean_target_adaptive}
\ell_{i,\mathrm{clean}}^{\mathrm{NFT\text{-}AW}}
=
\beta
\frac{
\left\|
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}^{\mathrm{NFT}}
\right\|^2
}{
\operatorname{sg}
\left[
\max
\left(
m\left(
\mathbf{x}_{0,i}^{\theta}
-
\mathbf{x}_{0,i}
\right),
\epsilon
\right)
\right]
}.
\end{equation}
Here the numerator still optimizes the reward-dependent NFT pseudo-target,
whereas the denominator measures the current clean-target reconstruction error
directly. Consequently, the adaptive scale depends on the sample, timestep,
and current model state, but is not distorted by the reward-dependent
displacement of the pseudo-target. In particular, it remains informative when
\(r_i\approx1/2\).





\subsubsection{Advantage Weighted Matching}
\label{sec:awm}

AWM  consider an on-policy update on a single sample drawn from
$\pi_{\theta}$. Let $\widehat{\pi}_{\theta}$ denote the ELBO-based surrogate of the sequence likelihood,
\begin{equation}
\nabla_{\theta}
\frac{\hat{\pi}_{\theta}(x_i^t \mid c)}
{\operatorname{sg}\left(\hat{\pi}_{\theta}(x_i^t \mid c)\right)}
\cdot A_i
=
\nabla_{\theta}\log \hat{\pi}_{\theta}(x_i^t \mid c)\cdot A_i
\approx
-\nabla_{\theta}
w_i  A_i
\left\|
\mathbf{v}^{\theta}_i-\mathbf{v}_i
\right\|^2,
\end{equation}
where $w_i=w(t)$ is a timestep-dependent weight. The local gradient-equivalent objective becomes
\begin{equation}
\label{eq:awm_velocity_target_short}
\ell_i^{\mathrm{AWM}}(\theta)
=
w_i A_i
\left\|
\mathbf{v}_i^\theta-\mathbf{v}_i
\right\|^2.
\end{equation}


Thus, AWM also fits Eq.~\ref{eq:unified_form}, with
\begin{equation}
\mathbf{\tau}_i
=
\mathbf{v}_i,
\qquad
\kappa_i
=
w_i A_i.
\end{equation}
AWM uses the flow matching velocity as its target. Reward does not move
the target itself; instead, the signed advantage changes the direction and
strength of the target. 
Positive-advantage samples pull the current
velocity toward the clean target, whereas negative-advantage samples push it
away by reversing the gradient of the squared-error term.



\textbf{Adding an old-velocity KL to AWM:}
\begin{equation}
\ell_i^{\mathrm{AWM+KL}}(\theta)
=
w_iA_i
\left\|
\mathbf{v}_i^\theta-\mathbf{v}_i
\right\|^2
+
\beta
\left\|
\mathbf{v}_i^\theta-\mathbf{v}_i^{\mathrm{old}}
\right\|^2 .
\end{equation}

Ignoring constants independent of $\theta$, objective can
be written as

\begin{equation}
\ell_i^{\mathrm{AWM+KL}}(\theta)
=
(w_iA_i+\beta)
\left\|
\mathbf{v}_i^\theta
\right\|^2
-
2
\left(\mathbf{v}_i^\theta\right)^\top
\left(
w_iA_i\mathbf{v}_i+\beta\mathbf{v}_i^{\mathrm{old}}
\right)
+
\mathrm{const.}
\end{equation}

When $w_iA_i+\beta>0$, completing the square gives
\begin{equation}
\ell_i^{\mathrm{AWM+KL}}(\theta)
=
(w_iA_i+\beta)
\left\|
\mathbf{v}_i^\theta
-
\widetilde{\mathbf{v}}_i^{\mathrm{AWM+KL}}
\right\|^2
+
\mathrm{const.},
\end{equation}
where
\begin{equation}
\widetilde{\mathbf{v}}_i^{\mathrm{AWM+KL}}
=
\frac{
w_iA_i\mathbf{v}_i+\beta\mathbf{v}_i^{\mathrm{old}}
}{
w_iA_i+\beta
} = \mathbf{v}_i^{\mathrm{old}} + \frac{w_iA_i}{w_iA_i+\beta} (\mathbf{v}_i-\mathbf{v}_i^{\mathrm{old}}) .
\end{equation}
