\section{Setup}
\label{sec:setup}
%------------------------------------------------------------
Let $c$ be a prompt and $x_1^{(1)},\dots,x_1^{(G)}$ a group of $G$ images sampled for it, with rewards
$R_i:=R(c,x_1^{(i)})$. We build it in two steps: pick a map
$f:\mathbb{R}\to\mathbb{R}$, apply it to each reward, and normalize inside the group
whenever the denominator is nonzero,
\begin{equation}\label{eq:weight-def}
w_i\;=\;\frac{f(R_i)}{\frac{1}{G}\sum_{j=1}^{G}f(R_j)},
\qquad\text{so that}\qquad
\frac{1}{G}\sum_{i=1}^{G}w_i=1 .
\end{equation}
The weights are nonnegative when $f(R_i)\ge0$ for every sample and at least
one mapped reward is positive.
Designing the algorithm is thus designing $f$, and only its shape
matters, since $f\mapsto af$ for $a>0$ cancels in \eqref{eq:weight-def}.

The default choice is $f(R)=\exp(R/\gamma)$, which makes \eqref{eq:weight-def} a group softmax,
\begin{equation}\label{eq:awr-group}
w_i\;=\;\frac{\exp(R_i/\gamma)}{\frac{1}{G}\sum_{j=1}^{G}\exp(R_j/\gamma)} .
\end{equation}
The
temperature $\gamma$ sets how concentrated the step is: $\gamma\to\infty$ gives $w_i\to1$, i.e.\ plain
supervised fine-tuning on the whole group, and $\gamma\to0$ puts everything on the best image, i.e.\
best-of-$N$.

\section{Variants of the Group Softmax}
\label{sec:variants}
%------------------------------------------------------------
Other constraints give other maps $f$,
$\bar{R}=\tfrac1G\sum_j R_j$ is the group mean and $z_i=(R_i-\bar{R})/\gamma$ the  reward, on
which Figure~\ref{fig:shapes} compares the shapes of $f$: they differ in how fast they grow, whether they reach zero at finite $z$.

\paragraph{Linear.}
\begin{equation}\label{eq:w-linear}
w_i\;=\;\frac{1+z_i}{\frac{1}{G}\sum_{j=1}^{G}(1+z_j)}
\;=\;1+z_i,
\end{equation}
since $z$ is centered, the normalizer is exactly $1$: the weight is linear in the centered reward,
with unit baseline at the group mean and no clipping. An image more than one temperature below the
mean ($z_i<-1$) therefore receives a negative weight: it is pushed away rather than ignored.

\paragraph{Tsallis-$q$.}
\begin{equation}\label{eq:w-tsallis}
w_i\;=\;\frac{\big[\,1+(q{-}1)z_i\,\big]_+^{1/(q-1)}}{\frac{1}{G}\sum_{j=1}^{G}\big[\,1+(q{-}1)z_j\,\big]_+^{1/(q-1)}},
\qquad q\in(1,2],
\end{equation}
a single knob from the softmax \eqref{eq:awr-group} at $q\to1$ to the clipped linear
weight $[1+z]_+$ at $q=2$, with the zero set moving in as $q$ grows: $w_i=0$ below $z_i=-1/(q-1)$.
For $q<2$ the truncation is what keeps the fractional power real; at $q=2$ the exponent is one,
the truncation is optional, and dropping it gives the linear weight \eqref{eq:w-linear}.

\paragraph{Sparsemax, from an $\alpha{=}2$ divergence.}
\begin{equation}\label{eq:w-sparsemax}
w_i\;=\;\frac{\big[\,R_i-\tau\,\big]_+}{\frac{1}{G}\sum_{j=1}^{G}\big[\,R_j-\tau\,\big]_+},
\end{equation}
where $\tau$ is a group quantile, e.g.\ the median. The temperature cancels in the normalization, so here
the threshold plays the role of $\gamma$: everything below $\tau$ gets $w_i=0$ and is dropped from the
step. Dropping the clip and setting $\tau=\bar{R}-\gamma$ gives $w_i=(R_i-\bar{R}+\gamma)/\gamma=1+z_i$,
the linear weight \eqref{eq:w-linear}.


\paragraph{Logistic, bounded.}
\begin{equation}\label{eq:w-logistic}
w_i\;=\;\frac{s(z_i)}{\frac{1}{G}\sum_{j=1}^{G}s(z_j)},\qquad s(z)=\frac{1}{1+e^{-z}},
\end{equation}
which saturates in both tails. Since $z$ is centered, the normalizer is close to $\tfrac12$ and hence
$w_i\lesssim2$: no image can take much more than $2/G$ of the step, however large its reward.

\paragraph{Rank, ordinal only.}
\begin{equation}\label{eq:w-rank}
w_i\;=\;\frac{\mathrm{rank}_i}{\frac{1}{G}\sum_{j=1}^{G}\mathrm{rank}_j}\;=\;\frac{2\,\mathrm{rank}_i}{G+1},
\qquad \mathrm{rank}_i\in\{1,\dots,G\},
\end{equation}
where the normalizer is the deterministic $(G{+}1)/2$, so no reward value ever enters the weight. Any
monotone recalibration of the reward model leaves the update unchanged.

\paragraph{Top-$k$, hard.}
\begin{equation}\label{eq:w-topk}
w_i\;=\;\frac{G}{k}\,\mathbbm{1}\big[\,R_i\in\mathrm{top}\text{-}k\,\big],
\end{equation}
already of unit mean without further normalization. The whole step comes from $k$ images: $k=1$ is
best-of-$N$, general $k$ is RAFT.

\paragraph{A worked example.}
Table~\ref{tab:numeric} runs one group of $G=8$ rewards through several of these weights. The group is
deliberately skewed: seven images spread over $[0.10,0.75]$ and one outlier at $R=1.75$, which sits $2.4$
standard deviations above the mean ($\bar{R}=0.625$, $\mathrm{std}(R)=0.463$). Two readings matter. Shape
decides concentration: with $\gamma=\mathrm{std}(R)$ the softmax hands $68\%$ of the step to one image out
of eight, the linear weight $43\%$, ranks only $22\%$ --- from the same rewards. And temperature matters
even more than shape: keeping the softmax but dividing $\gamma$ by three leaves $99.7\%$ of the step on
the outlier, i.e.\ best-of-$N$ in all but name. Since the weights multiply the regression target rather
than a gradient, how spread out they are inside a group is also how much variance the regression cannot
remove, which is why we keep $\gamma$ at the scale of $\mathrm{std}(R)$ and winsorize outliers.

\paragraph{Every variant is a reward relabeling.}
Any positive $f$ can be written as $f(R)=\exp(\tilde{R}/\gamma)$ with $\tilde{R}=\gamma\log f(R)$, so
picking a variant is the same as keeping the softmax \eqref{eq:awr-group} and feeding it a relabeled
reward: dividing by $\mathrm{std}(R)$ is an adaptive temperature, $\gamma\log(2\,\mathrm{rank}/(G{+}1))$
turns \eqref{eq:awr-group} into \eqref{eq:w-rank}, $\min(R,q_{0.95})$ is the winsorized column of
Table~\ref{tab:numeric}, and summing several rewards as $\sum_k R_k/s_k$ multiplies their softmaxes, so
one failed reward suppresses the image whatever the others say (an AND), with each $s_k$ acting as that
reward's own temperature. Averaging weights across rewards instead allows compensation (an OR). The
linear weight \eqref{eq:w-linear} is the exception: it changes sign at $z=-1$, and no relabeled
softmax ever returns a negative weight.

\section{Temperature Proof}
\label{sec:temperature-limits}

Equation~(2) defines the group softmax weights
\[
w_i=\frac{\exp(R_i/\gamma)}{\frac{1}{G}\sum_{j=1}^{G}\exp(R_j/\gamma)} .
\]
This section proves the two statements made below Equation~(2). In both proofs the group size $G$ and the rewards $R_1,\dots,R_G$ are fixed, $G$ is finite, and the temperature $\gamma>0$ is the varying quantity.

\subsection{Infinite temperature: $w_i \to 1$}

We prove that $w_i\to 1$ for every image $i$ as $\gamma\to\infty$.

\begin{enumerate}
\item Fix an image $i$. Its reward $R_i$ is a fixed number and $\gamma\to\infty$, so
\[
\frac{R_i}{\gamma}\to 0 .
\]
\item The exponential function is continuous, therefore
\[
\exp\left(\frac{R_i}{\gamma}\right)\to\exp(0)=1 .
\]
The same holds for every image $j\in\{1,\dots,G\}$, so each of the $G$ terms in the denominator of $w_i$ tends to $1$.
\item The denominator is the average of these $G$ terms. The number of terms $G$ is fixed, and each term tends to $1$, so
\[
\frac{1}{G}\sum_{j=1}^{G}\exp\left(\frac{R_j}{\gamma}\right)\to\frac{1}{G}\cdot G\cdot 1=1 .
\]
\item By step~2 the numerator of $w_i$ tends to $1$, and by step~3 the denominator tends to $1\neq 0$. The limit of a fraction whose numerator and denominator both converge is the quotient of the limits whenever the limit of the denominator is not zero, hence
\[
w_i\to\frac{1}{1}=1 .
\]
\end{enumerate}

The share of image $i$ in the step is $w_i/G$, so every share tends to $1/G$: all $G$ images enter the update with equal weight. The step is the unweighted average of the per-image updates, i.e.\ plain supervised fine-tuning on the whole group.

\subsection{Zero temperature: everything on the best image}

Let $M=\max_j R_j$ and let $i^\star$ be the image with $R_{i^\star}=M$. Assume first that $i^\star$ is unique; the case of several images attaining $M$ is covered by the remark at the end. We prove that the step is carried entirely by image $i^\star$ as $\gamma\to 0$.

\begin{enumerate}
\item For every $\gamma>0$ the number $\exp(-M/\gamma)$ is positive, so multiplying the numerator and the denominator of $w_i$ by it does not change the value of the fraction. Using $\exp(a)\exp(b)=\exp(a+b)$ in both the numerator and the denominator,
\[
w_i
=\frac{\exp(R_i/\gamma)\,\exp(-M/\gamma)}
{\left(\frac{1}{G}\sum_{j=1}^{G}\exp(R_j/\gamma)\right)\exp(-M/\gamma)}
=\frac{\exp\left(\frac{R_i-M}{\gamma}\right)}
{\frac{1}{G}\sum_{j=1}^{G}\exp\left(\frac{R_j-M}{\gamma}\right)} .
\]
\item By the definition of $M$ we have $R_i-M\le 0$ for every image, and $R_i-M<0$ for every image other than $i^\star$.
\item Fix an image $i\neq i^\star$. The number $R_i-M$ is fixed and negative, so
\[
\frac{R_i-M}{\gamma}\to-\infty \qquad(\gamma\to 0),
\]
and since $\exp(x)\to 0$ for $x\to-\infty$,
\[
\exp\left(\frac{R_i-M}{\gamma}\right)\to 0 .
\]
\item For the best image, $R_{i^\star}-M=0$, hence $\frac{R_{i^\star}-M}{\gamma}=0$ for every $\gamma$, and
\[
\exp\left(\frac{R_{i^\star}-M}{\gamma}\right)=1 \quad\text{for every }\gamma .
\]
\item The denominator of the rewritten $w_i$ is the average of $G$ terms. By step~3, the $G-1$ terms with $i\neq i^\star$ tend to $0$, and by step~4 the term with $i^\star$ is equal to $1$ for every $\gamma$. Therefore
\[
\frac{1}{G}\sum_{j=1}^{G}\exp\left(\frac{R_j-M}{\gamma}\right)\to\frac{1}{G}\cdot 1=\frac{1}{G} .
\]
\item For the best image the numerator is $1$ for every $\gamma$ (step~4) and the denominator tends to $1/G\neq 0$ (step~5), so
\[
w_{i^\star}\to\frac{1}{1/G}=G .
\]
\item For any other image the numerator tends to $0$ (step~3) and the denominator tends to $1/G\neq 0$ (step~5), so
\[
w_i\to\frac{0}{1/G}=0 .
\]
\item The share of the best image in the step is $w_{i^\star}/G$, which tends to $G/G=1$, and every other share tends to $0$. The step is carried entirely by the best image, i.e.\ best-of-N.
\end{enumerate}






\section{An Exploration of Softmax and Its Variants}
\begin{algorithm}[H]
    \caption{Single Target Loss Exploration of Softmax Scheme}
    \label{alg:velocity-add-delete}
    \textbf{Require:} Initial model $\theta_0$, reward function $R$, prompt distribution $p(c)$, number of prompts per iteration $C$, rollout size per prompt $K$, batch size $B$, total samples per iteration $D = KC$, rollout coverage coefficient $\beta$, mass-shift clip $\Delta_{\mathrm{clip}}$, reference regularization $\lambda_{\mathrm{ref}}$, EMA rate $\eta$, reward-scale constant $\epsilon_R$, numerical constant $\epsilon_{\mathrm{alg}}$, gradient accumulation steps $N_{\mathrm{acc}}$.\\
    \textbf{Initialize:} Training policy $\theta \leftarrow \theta_0$, old policy $\theta_{\mathrm{old}} \leftarrow \theta_0$, reference policy $\theta_{\mathrm{ref}} \leftarrow \theta_0$, data buffer $\mathcal{D} \leftarrow \emptyset$.
    \begin{algorithmic}[1]
        \For{\text{each iteration}}
            \For{\text{each prompt $c_j \sim p(c)$, $j = 1, \dots, C$}} \Comment{Rollout Step, Data Collection}
                \State Sample $K$ clean images $\{\mathbf{x}_0^{j,k}\}_{k=1}^{K}$ with $\theta$ conditioned on $c_j$.
                \State Evaluate rewards $R_{j,k} \leftarrow R(\mathbf{x}_0^{j,k}, c_j)$ for $k = 1, \dots, K$.
            \EndFor

            % \State Compute prompt-wise reward means
            % $\bar{R}_{c_j} \leftarrow \frac{1}{K}\sum_{k=1}^{K} R_{j,k}$ for $j = 1, \dots, C$.

            % \State Compute global reward mean
            % $\bar{R} \leftarrow \frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K} R_{j,k}$.

            % \State Compute global reward scale
            % $\sigma_R \leftarrow \sqrt{\frac{1}{D}\sum_{j=1}^{C}\sum_{k=1}^{K}(R_{j,k} - \bar{R})^2}.$

            \For{\text{each prompt $c_j$, $j = 1, \dots, C$}}
                \State Compute normalized weights
                \[
                    w_{j,k} \leftarrow \frac{f(R_{j,k})}{\frac{1}{G}\sum_{k=1}^{G}f(R_{j,k})},
                    \quad k = 1, \dots, K.
                \]

                \State Store
                $\mathcal{D}\leftarrow\mathcal{D}\cup
                \{(c_j,\mathbf{x}_0^{j,k},w_{j,k})\}_{k=1}^{K}$.
            \EndFor

            \For{\text{each mini batch }
            $\{(c_i,\mathbf{x}_i^0,w_i)\}_{i=1}^{B}\subset\mathcal{D}$}
                \State Sample $t_i\sim\mathcal{U}(0,1)$,
                $\boldsymbol{\epsilon}_i\sim\mathcal{N}(\mathbf{0},\mathbf{I})$;
                set $\mathbf{x}_i^{t_i}\leftarrow
                (1-t_i)\mathbf{x}_i^0+t_i\boldsymbol{\epsilon}_i$.
                \State Predict $\mathbf{x}_i^\theta$,
                $\mathbf{x}_i^{\mathrm{old}}$, and $\mathbf{x}_i^{\mathrm{ref}}$
                from $(\mathbf{x}_i^{t_i},c_i,t_i)$ with
                $\theta$, $\theta_{\mathrm{old}}$, and $\theta_{\mathrm{ref}}$.
                \State $\mathbf{x}_i^{\mathrm{target}}\leftarrow
                \mathbf{x}_i^{\mathrm{old}}+
                \beta(w_i-1)(\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}})$.
                \State Compute
                \[
                    \begin{aligned}
                    \mathcal{L}_{\mathrm{policy}}
                    &\leftarrow\frac{1}{B}\sum_{i=1}^{B}
                    \frac{\|\mathbf{x}_i^\theta-\mathbf{x}_i^{\mathrm{target}}\|_2^2}
                    {\operatorname{sg}[\operatorname{mean}
                    |\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}}|]},\\
                    \mathcal{L}_{\mathrm{ref}}
                    &\leftarrow\frac{1}{B}\sum_{i=1}^{B}
                    \|\mathbf{v}_i^\theta-\mathbf{v}_i^{\mathrm{ref}}\|_2^2.
                    \end{aligned}
                \]
                \State Backpropagate
                $(\Delta_{clip}\mathcal{L}_{\mathrm{policy}}+
                \lambda_{\mathrm{ref}}\mathcal{L}_{\mathrm{ref}})/N_{\mathrm{acc}}$.
            \EndFor
            \State Update $\theta$; set
            $\theta_{\mathrm{old}}\leftarrow
            \eta\theta_{\mathrm{old}}+(1-\eta)\theta$ and
            $\mathcal{D}\leftarrow\emptyset$.
        \EndFor
    \end{algorithmic}
    \textbf{Output:} $\mathbf{v}_\theta$
\end{algorithm}

For each prompt \(c_j\), let \(G=K\) and define
\[
\bar R_{c_j}=\frac{1}{G}\sum_{i=1}^{G}R_{j,i},
\qquad
\sigma_{c_j}=\sqrt{\frac{1}{G}\sum_{i=1}^{G}
\bigl(R_{j,i}-\bar R_{c_j}\bigr)^2}.
\]
Unless stated otherwise, each experiment applies the within-prompt normalization
in \eqref{eq:weight-def}, so \(G^{-1}\sum_i w_{j,i}=1\).



\paragraph{Motivation and experiment plan.}
We want to understand why the softmax weights in FlowAWR can lead to fast
convergence. Under \eqref{eq:weight-def}, they give more of each group's total
weight to samples with higher rewards. We test whether this focus on better
samples is the main reason.

Experiment A pushes this idea to the extreme situation. A one-hot rule gives all weight to
the top sample(s) and zero to the rest. This sharp choice may make training
unstable, so we also give weight to samples within \(cR_j^{\max}\) of the top
reward, without using the group standard deviation for this threshold.
If no reward in the group is positive, we use uniform weights and apply no
reward-driven target shift to that group.

Experiment B straightly tests the idea with softmax. Compared with \(c=1\),
\(c<1\) puts more weight on high-reward samples, while \(c>1\) spreads weight
more evenly. If \(c<1\) works better, it supports our idea.

Experiment C uses the unshifted linear mapping as a baseline. Experiment D
compares sigmoid with linear within each group: depending on \(c\) and the
rewards, sigmoid can yield either sharper or flatter normalized weights.
We test whether smaller \(c\) helps, whether \(c=0.3\) underperforms, and
whether a small-\(c\) sigmoid is preferable to the softmax-style mappings.

Experiment E changes the power \(p\) to
test whether steeper or flatter maps work better. It is another view to test our idea.

Experiment F uses an L1
scale. Its weight curve looks much like the softmax curve in Exp.B, so it is
less important and may be skipped.



\paragraph{Experiment A: One-hot and near-maximum threshold.}

Let
\[
R_j^{\max}=\max_i R_{j,i},\qquad
T_j(c)=R_j^{\max}-cR_j^{\max}=(1-c)R_j^{\max}.
\]
\[
S_j(c)=\{i:R_{j,i}\ge T_j(c)\}\qquad\text{for }R_j^{\max}>0.
\]
Set
\[
w_{j,i}(c)
\;=\;
\begin{cases}
\displaystyle\frac{G}{|S_j(c)|}\,\mathbbm{1}[i\in S_j(c)],
& R_j^{\max}>0,\\[6pt]
1, & R_j^{\max}\le0,
\end{cases}
\qquad c\in\{0,0.05,0.1,0.25\}.
\]
For \(R_j^{\max}>0\), when the maximum is unique, \(c=0\) is one-hot with weight \(G\) on the
max-reward sample. If several samples are equal-reward at the maximum, they share the
G mass equally. Increasing \(c\) includes samples
within a larger fraction of the maximum reward. 

For \(R_j^{\max}\le0\),
all $w_{j,i}(c)$ equal \(1\), so the probability for each sample in this group $c_j$ will not change. While other loss terms like mse between $v_{\theta}$ and $v_{old}$ and KL terms still apply.

\textbf{Target}: Test whether assigning most or all of each group's weight
mass to its best-reward samples is the main source of the gain from
softmax weighting.

% \(\tau_j=R_{j,(k)}\),
% \(n_{>,j}=|\{i:R_{j,i}>\tau_j\}|\), and
% \(n_{=,j}=|\{i:R_{j,i}=\tau_j\}|\). The weights are
% \[
% w_{j,i}^{\mathrm{top}\text{-}k}=\begin{cases}
% G/k, & R_{j,i}>\tau_j,\\[3pt]
% (G/k)(k-n_{>,j})/n_{=,j}, & R_{j,i}=\tau_j,\\[3pt]
% 0, & R_{j,i}<\tau_j.
% \end{cases}
% \]
% For \(G=24\), use \(k\in\{1,3,6,12\}\). With distinct rewards this is
% hard Top-$k$; only a tie at the boundary receives fractional weights. Both
% rules satisfy \(\sum_i w_{j,i}=G\). Compare performance across \(c\) and
% \(k\) to test how much concentration on the best samples helps.

\paragraph{Experiment B: Softmax with standard-deviation temperature.}

\[
f_c(R_{j,i})=\exp\!\left(\frac{R_{j,i}}{c\sigma_{c_j}}\right),
\qquad c\in\{0.5,0.75,1.0,1.5\}.
\]
For \(\sigma_{c_j}>0\), smaller \(c\) yields a steeper mapping and more
concentrated weights; larger \(c\) gives a flatter mapping.

\textbf{Target}: This exp tests
the trade-off between concentrating on high rewards and keeping a flatter
training signal. We want to test whether more concentrating or flat than classical softmax is better for optimize.

\paragraph{Experiment C: Linear baseline.}
\[
f_{\mathrm{lin}}(R)=[R]_+,
\qquad
f_{\mathrm{lin,shift}}(R)=[1+R]_+,
\qquad [x]_+=\max(x,0).
\]
Only \(f_{\mathrm{lin}}(R)=[R]_+\) is included in the planned sweep;
the shifted mapping is shown for comparison, not run as an experiment.

\textbf{Target}: Linear reward mapping acts as baseline variant. Expect to be worse than softmax and one-hot.

\paragraph{Experiment D: Sigmoid mapping.}
\[
f_{\mathrm{sig},c}(R_{j,i})
=\frac{1}{1+\exp(-(R_{j,i}-0.5)/c)},
\qquad c\in\{0.01,0.1,0.3\}.
\]

The smaller \(c\) is, the sharper the transition around \(R=0.5\).
After within-group normalization, the sigmoid may be sharper or flatter
than linear, depending on the reward range; its saturation also limits
how much the highest rewards can dominate.

\textbf{Target}: Compare sigmoid with the linear baseline within each group.
Test whether smaller \(c\) improves performance, whether the flatter
\(c=0.3\) setting underperforms, and whether a small-\(c\) sigmoid is a
better choice than the softmax-style mappings in Experiment B.


\paragraph{Experiment E: Power mapping.}

\[
f_p(R_{j,i})=\bigl[R_{j,i}\bigr]_+^p.
\]

We condider \(p\in\{0.5,2.0\}\).

\textbf{Target}: Compare with the linear \(p=1\) to test
whether the stronger emphasis on high rewards gives better behavior (similar
performance with softmax).

\paragraph{Experiment F: Softmax with L1 scale.}
Define
\[
||R_{j,i}-\bar{R}_{c_j}||_1=\frac{1}{G}\sum_{i=1}^{G}
\bigl|R_{j,i}-\bar R_{c_j}\bigr|.
\]
Then
\[
f_{c,1}(R_{j,i})
=\exp\!\left(\frac{R_{j,i}}{c||R_{j,i}-\bar{R}_{c_j}||_1}\right),
\qquad c\in\{0.5,0.75,1.0,1.5\}.
\]


\textbf{Target}: This is a variant for softmax. It is similiar with softmax with std temperature as shown in following figure. So we expect a similar performance with Experiment B. This exp can be skipped if nessesary.

\paragraph{Experiment G: Exponential on positive excess with mass-preserving rescaling.}
As in Experiment C, we use the linear mapping
\[
w_{j,i}=
\begin{cases}
\displaystyle
\frac{[R_{j,i}]_+}{\frac{1}{G}\sum_{k=1}^{G}[R_{j,k}]_+},
& \displaystyle \sum_{k=1}^{G}[R_{j,k}]_+>0,\\[10pt]
1, & \displaystyle \sum_{k=1}^{G}[R_{j,k}]_+=0,
\end{cases}
\]
so that \(\frac{1}{G}\sum_{i=1}^{G}w_{j,i}=1\). We
apply an exponential only to the positive excess, while leaving the
non-positive excess unchanged:
\[
h_{j,i}(\alpha)
=
\begin{cases}
\exp\!\bigl(\alpha\, (w_{j,i}-1)\bigr), & w_{j,i}-1>0,\\[4pt]
w_{j,i}-1, & w_{j,i}-1\le 0,
\end{cases}
\qquad
\alpha\in\{1.0,5.0,10.0,20.0\}.
\]
Because the exponential changes the total positive mass, we rescale the
positive part so that its sum is preserved:
\[
S_j^{+}=\sum_{i:\,w_{j,i}-1>0} (w_{j,i}-1),
\qquad
\widetilde{S}_j^{+}(\alpha)=\sum_{i:\,w_{j,i}-1>0} h_{j,i}(\alpha),
\]
\[
\tilde w_{j,i}
=
\begin{cases}
\displaystyle
\frac{S_j^{+}}{\widetilde{S}_j^{+}(\alpha)}\, h_{j,i}(\alpha),
& w_{j,i}-1>0,\\[10pt]
w_{j,i}-1, & w_{j,i}-1\le 0.
\end{cases}
\]
Since \(\sum_{i:\,w_{j,i}-1>0}\tilde w_{j,i}=S_j^{+}\) and the
negative part is untouched, we still have
\(\sum_{i=1}^{G}\tilde w_{j,i}=0\). The target shift then uses
the same form as the algorithm:
\[
\mathbf{x}_i^{\mathrm{target}}
=
\mathbf{x}_i^{\mathrm{old}}
+
\beta\,\tilde w_{j,i}(\mathbf{x}_i^0-\mathbf{x}_i^{\mathrm{old}}).
\]
